function buildPrompt(roomType: string, style: string, tier: string, budget: number) {
  const styleConfig = styleVocabulary[style];
  
  if (!styleConfig) {
    throw new Error(`Ukendt stil: ${style}`);
  }
  
  const tierConfig = styleConfig[tier];
  
  // VIKTIG: Start prompt med stil for at tvinge den
  const stylePrefix = {
    'skandinavisk': 'SCANDINAVIAN STYLE: white walls, light wood, minimal, airy, natural light',
    'moderne': 'MODERN STYLE: clean lines, neutral colors, functional, contemporary',
    'badboy': 'DARK MASCULINE LUXURY STYLE: MATTE BLACK WALLS, leather, chrome, moody lighting, NO WHITE, NO LIGHT WOOD, NO SCANDINAVIAN ELEMENTS'
  }[style];
  
  const prompt = `
    ${stylePrefix}
    
    ${tierConfig.prompt}
    
    CRITICAL: This must be ${style} style ONLY. 
    DO NOT use scandinavian elements unless requested.
    DO NOT default to white walls.
    
    Transform this ${roomType}.
    Maintain exact room structure.
  `;
  
  return prompt;
}
