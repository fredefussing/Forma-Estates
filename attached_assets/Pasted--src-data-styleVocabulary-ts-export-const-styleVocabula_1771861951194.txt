// /server/routes.ts

function buildPrompt(roomType: string, style: string, tier: string, budget: number) {
  const styleConfig = styleVocabulary[style];
  const tierConfig = styleConfig[tier];
  
  // VIKTIG: Specificer vægfarve eksplicit
  const wallColorInstruction = style === 'badboy' 
    ? "MATTE BLACK WALLS mandatory, dark charcoal surfaces, NO white walls, NO light colors"
    : "white walls, light surfaces";
  
  const prompt = `
    Transform this ${roomType} to ${style} style.
    ${tierConfig.prompt}
    
    CRITICAL WALL COLOR: ${wallColorInstruction}
    
    Style elements: ${styleConfig.colors.join(", ")}.
    Materials: ${styleConfig.materials.join(", ")}.
    Furniture: ${styleConfig.furniture.join(", ")}.
    Lighting: ${styleConfig.lighting.join(", ")}.
    
    Maintain exact room structure, windows, doors, NO layout changes.
    Photorealistic, high quality, 4K render.
  `;
  
  return prompt;
}

// /src/components/StyleSelector.tsx

const styles = [
  { id: 'skandinavisk', name: 'Skandinavisk', icon: '🌿' },
  { id: 'moderne', name: 'Moderne', icon: '◼️' },
  { id: 'badboy', name: 'Badboy (New Money)', icon: '🖤' },
  // MIDTÅRHUNDREDE FJERNET
];

export function StyleSelector({ onChange }) {
  return (
    <div className="style-selector">
      <h3>Vælg stil</h3>
      <div className="style-grid">
        {styles.map(style => (
          <button 
            key={style.id}
            onClick={() => onChange(style.id)}
            className={`style-card ${style.id === 'badboy' ? 'dark-style' : ''}`}
          >
            <span className="icon">{style.icon}</span>
            <span className="name">{style.name}</span>
            {style.id === 'badboy' && (
              <span className="badge">Mørk</span>
            )}
          </button>
        ))}
      </div>
    </div>
  );
}
