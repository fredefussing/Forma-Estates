export function QuoteRequest({ 
  designId, 
  generatedImageUrl, 
  roomType, 
  style, 
  budget 
}) {
  const [email, setEmail] = useState('');
  const [notes, setNotes] = useState('');
  const [submitted, setSubmitted] = useState(false);
  
  if (submitted) {
    return (
      <div className="confirmation">
        <h3>✓ Din forespørgsel er sendt!</h3>
        <p>Vi kontakter dig indenfor 24 timer med personligt tilbud.</p>
      </div>
    );
  }
  
  return (
    <div className="quote-request">
      <h3>Er du glad for dit design?</h3>
      <p>Få et tilbud på at gøre det til virkelighed.</p>
      
      <input
        type="email"
        placeholder="din@email.dk"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
      />
      
      <textarea
        placeholder="Dine ønsker eller ændringer (max 100 tegn)..."
        maxLength={100}
        value={notes}
        onChange={(e) => setNotes(e.target.value)}
      />
      <span>{notes.length}/100</span>
      
      <button 
        onClick={() => {
          fetch('/api/quote-request', {
            method: 'POST',
            body: JSON.stringify({
              designId,
              email,
              notes,
              generatedImageUrl,
              roomType,
              style,
              budget
            })
          });
          setSubmitted(true);
        }}
        disabled={!email}
      >
        Få tilbud
      </button>
    </div>
  );
}
