5. GSAP Zoom + Før/Efter + 360° Parallax
tsx
Copy
// Når brugeren klikker på et rum:

const handleRoomClick = (room: Room) => {
  // 1. GSAP zoom fra plantegning til rum
  gsap.to(floorplanRef.current, {
    scale: 3,
    x: calculateXOffset(room),
    y: calculateYOffset(room),
    duration: 1.2,
    ease: "power3.inOut",
    onComplete: () => {
      // 2. Skift til før/efter view
      setViewMode("beforeAfter");
      
      // 3. Start Ken Burns + Parallax
      startKenBurns();
    }
  });
};

// Før/Efter med Ken Burns + Parallax:
<div 
  className="relative overflow-hidden"
  onMouseMove={handleParallax}
>
  {/* Efter-billede (baggrund) med Ken Burns */}
  <img 
    src={afterUrl}
    className="absolute inset-0 w-full h-full object-cover ken-burns"
  />
  
  {/* Før-billede (top, med clip) */}
  <img 
    src={beforeUrl}
    className="absolute inset-0 w-full h-full object-cover"
    style={{ clipPath: `inset(0 ${100 - reveal}% 0 0)` }}
  />
  
  {/* Slider handle */}
  <div className="slider-handle" style={{ left: `${reveal}%` }} />
</div>

// Parallax handler:
const handleParallax = (e: React.MouseEvent) => {
  const rect = e.currentTarget.getBoundingClientRect();
  const x = (e.clientX - rect.left) / rect.width - 0.5;
  const y = (e.clientY - rect.top) / rect.height - 0.5;
  
  gsap.to(afterImageRef.current, {
    x: -x * 30,   // 30px max bevægelse
    y: -y * 20,   // 20px max bevægelse
    duration: 0.5,
    ease: "power2.out",
  });
};

// db/schema.ts — tjek at disse felter findes:

export const properties = mysqlTable("properties", {
  id: int("id").autoincrement().primaryKey(),
  name: varchar("name", { length: 255 }).notNull(),
  floorplanUrl: text("floorplan_url"),        // Almindelig plantegning
  floorplan3dUrl: text("floorplan_3d_url"),   // AI-genereret 3D plantegning
  style: varchar("style", { length: 100 }),   // Valgt stil
  tier: varchar("tier", { length: 50 }),      // Valgt kvalitet
  status: varchar("status", { length: 50 }).default("draft"),
  // === NYT: Plantegnings-analyse ===
  floorplanAnalysis: text("floorplan_analysis"), // JSON med rum-data
  createdAt: timestamp("created_at").defaultNow(),
});

export const propertyRooms = mysqlTable("property_rooms", {
  id: int("id").autoincrement().primaryKey(),
  propertyId: int("property_id").notNull(),
  name: varchar("name", { length: 255 }).notNull(),
  roomType: varchar("room_type", { length: 100 }), // "living_room", "bedroom"
  posX: float("pos_x").default(0),           // % position på plantegning
  posY: float("pos_y").default(0),
  width: float("width").default(20),         // % størrelse på plantegning
  height: float("height").default(20),
  color: varchar("color", { length: 50 }).default("#8B9EB7"),
  hasImages: int("has_images").default(0),
  // === NYT: Plantegnings-analyse per rum ===
  analysisData: text("analysis_data"),        // JSON med vinduer, døre, etc.
  createdAt: timestamp("created_at").defaultNow(),
});

export const roomImages = mysqlTable("room_images", {
  id: int("id").autoincrement().primaryKey(),
  propertyRoomId: int("property_room_id").notNull(),
  beforeUrl: text("before_url"),              // Før-billede (uploadet)
  afterUrl: text("after_url"),                // Efter-billede (AI-genereret)
  angle: varchar("angle", { length: 100 }).default("center"),
  status: varchar("status", { length: 50 }).default("pending"),
  createdAt: timestamp("created_at").defaultNow(),
});
8. Komplet flow — API kald rækkefølge
plain
Copy
Bruger:    Upload plantegning
           ↓
Frontend:  trpc.property.create.mutate({ name, floorplanUrl })
           ↓
Backend:   Opret property i DB → return { propertyId }
           ↓
Frontend:  trpc.generate.floorplan3d.mutate({ 
             prompt: name + room descriptions,
             aspectRatio: "16:9" 
           })
           ↓
Backend:   fal.ai nano-banana-2 → 3D plantegning
           Gem floorplan3dUrl i DB
           ↓
Frontend:  trpc.generate.analyzeFloorplan.mutate({
             floorplanUrl,
             propertyId
           })
           ↓
Backend:   AI analyserer plantegning → rum-data
           Gem analysis i DB
           ↓
Bruger:    Vælg rum + upload før-billeder
           ↓
Frontend:  trpc.property.addImage.mutate({
             propertyRoomId, beforeUrl, angle
           })
           ↓
Bruger:    Vælg stil + tier → Klik "Generér"
           ↓
Frontend:  // PARALLELT for alle rum:
           Promise.all(rooms.map(room => 
             trpc.generate.afterImage.mutate({
               beforeImageUrl: room.beforeUrl,
               roomName: room.name,
               style: "skandinavisk",
               tier: "premium",
               posX: room.posX,     // Plantegnings-kontekst!
               posY: room.posY,
               width: room.width,
               height: room.height,
               allRooms: allRooms,   // Naborum info!
             })
           ))
           ↓
Backend:   Collov AI med STRICT PRESERVATION prompt
           + layout context fra plantegning
           → Gem afterUrl i DB
           ↓
Frontend:  Fremvisning! 
           Plantegning med klikbare rum
           → Zoom → Før/Efter + 360° parallax
           → "Lav video" knap
