import { useRef, useState } from "react";
import { Canvas, useFrame } from "@react-three/fiber";
import { OrbitControls, Text } from "@react-three/drei";
import * as THREE from "three";

interface Room {
  id: number;
  name: string;
  x: number;
  y: number;
  width: number;
  height: number;
  color: string;
  style?: string | null;
  generatedImageUrl?: string | null;
  description?: string | null;
}

interface House3DProps {
  rooms: Room[];
  onRoomClick: (room: Room) => void;
}

function RoomBlock({ room, onClick, isHovered, setHovered }: {
  room: Room;
  onClick: () => void;
  isHovered: boolean;
  setHovered: (v: boolean) => void;
}) {
  const meshRef = useRef<THREE.Mesh>(null);

  useFrame(() => {
    if (!meshRef.current) return;
    const targetS = isHovered ? 1.05 : 1;
    const targetH = isHovered ? 0.3 : 0.05;
    meshRef.current.scale.x += (targetS - meshRef.current.scale.x) * 0.1;
    meshRef.current.scale.y += (targetS - meshRef.current.scale.y) * 0.1;
    meshRef.current.scale.z += (targetS - meshRef.current.scale.z) * 0.1;
    meshRef.current.position.y += (targetH - meshRef.current.position.y) * 0.1;
  });

  const width = room.width;
  const depth = room.height;
  const posX = room.x + width / 2;
  const posZ = room.y + depth / 2;

  return (
    <group position={[posX, 0, posZ]}>
      <mesh
        ref={meshRef}
        position={[0, 0.05, 0]}
        rotation={[-Math.PI / 2, 0, 0]}
        onClick={(e) => { e.stopPropagation(); onClick(); }}
        onPointerOver={(e) => { e.stopPropagation(); setHovered(true); document.body.style.cursor = "pointer"; }}
        onPointerOut={() => { setHovered(false); document.body.style.cursor = "auto"; }}
      >
        <planeGeometry args={[width - 0.1, depth - 0.1]} />
        <meshStandardMaterial color={room.color} transparent opacity={isHovered ? 0.95 : 0.75} side={THREE.DoubleSide} />
      </mesh>

      <group>
        <mesh position={[0, 0.25, -depth / 2 + 0.05]}>
          <boxGeometry args={[width, 0.5, 0.05]} />
          <meshStandardMaterial color={room.color} opacity={0.3} transparent />
        </mesh>
        <mesh position={[0, 0.25, depth / 2 - 0.05]}>
          <boxGeometry args={[width, 0.5, 0.05]} />
          <meshStandardMaterial color={room.color} opacity={0.3} transparent />
        </mesh>
        <mesh position={[-width / 2 + 0.05, 0.25, 0]}>
          <boxGeometry args={[0.05, 0.5, depth]} />
          <meshStandardMaterial color={room.color} opacity={0.3} transparent />
        </mesh>
        <mesh position={[width / 2 - 0.05, 0.25, 0]}>
          <boxGeometry args={[0.05, 0.5, depth]} />
          <meshStandardMaterial color={room.color} opacity={0.3} transparent />
        </mesh>
      </group>

      <Text position={[0, 0.55, 0]} fontSize={0.25} color="#F5F2ED" anchorX="center" anchorY="middle">
        {room.name}
      </Text>

      {room.generatedImageUrl && (
        <mesh position={[width / 2 - 0.3, 0.6, -depth / 2 + 0.3]}>
          <sphereGeometry args={[0.12, 8, 8]} />
          <meshStandardMaterial color="#4A6741" emissive="#4A6741" emissiveIntensity={0.5} />
        </mesh>
      )}
    </group>
  );
}

function FloorBase({ rooms }: { rooms: Room[] }) {
  const maxX = Math.max(...rooms.map((r) => r.x + r.width)) + 1;
  const maxZ = Math.max(...rooms.map((r) => r.y + r.height)) + 1;
  return (
    <mesh rotation={[-Math.PI / 2, 0, 0]} position={[maxX / 2 - 0.5, 0, maxZ / 2 - 0.5]}>
      <planeGeometry args={[maxX + 2, maxZ + 2]} />
      <meshStandardMaterial color="#1A1A1A" />
    </mesh>
  );
}

function Scene({ rooms, onRoomClick }: House3DProps) {
  const [hoveredRoom, setHoveredRoom] = useState<number | null>(null);
  const maxX = Math.max(...rooms.map((r) => r.x + r.width)) + 1;
  const maxZ = Math.max(...rooms.map((r) => r.y + r.height)) + 1;
  const centerX = maxX / 2 - 0.5;
  const centerZ = maxZ / 2 - 0.5;

  return (
    <>
      <ambientLight intensity={0.4} />
      <directionalLight position={[5, 8, 5]} intensity={0.8} />
      <pointLight position={[centerX, 3, centerZ]} intensity={0.3} />
      <FloorBase rooms={rooms} />
      {rooms.map((room) => (
        <RoomBlock key={room.id} room={room} onClick={() => onRoomClick(room)} isHovered={hoveredRoom === room.id} setHovered={(v) => setHoveredRoom(v ? room.id : null)} />
      ))}
      <OrbitControls enablePan={true} enableZoom={true} enableRotate={true} maxPolarAngle={Math.PI / 2.2} minPolarAngle={0.2} minDistance={3} maxDistance={20} target={[centerX, 0, centerZ]} />
    </>
  );
}

export default function House3D({ rooms, onRoomClick }: House3DProps) {
  const maxX = Math.max(...rooms.map((r) => r.x + r.width)) + 1;
  const maxZ = Math.max(...rooms.map((r) => r.y + r.height)) + 1;
  const centerX = maxX / 2 - 0.5;
  const centerZ = maxZ / 2 - 0.5;

  return (
    <div className="w-full h-full rounded-xl overflow-hidden bg-[#0C0C0C]">
      <Canvas camera={{ position: [centerX, 10, centerZ + 5], fov: 45 }} style={{ width: "100%", height: "100%" }}>
        <Scene rooms={rooms} onRoomClick={onRoomClick} />
      </Canvas>
    </div>
  );
}

import { useState } from "react";
import { Button } from "@/components/ui/button";
import { trpc } from "@/providers/trpc";
import { X, Sparkles, Loader2 } from "lucide-react";

const STYLES = [
  { id: "moderne", name: "Moderne", icon: "🏠" },
  { id: "klassisk", name: "Klassisk", icon: "🛋️" },
  { id: "nordisk", name: "Nordisk", icon: "🌲" },
  { id: "kyst", name: "Kyst", icon: "🌊" },
  { id: "luksus", name: "Luksus", icon: "💎" },
];

interface Room {
  id: number;
  name: string;
  color: string;
  images: { angle: string; beforeUrl: string; afterUrl?: string }[];
}

interface RoomModalProps {
  room: Room;
  onClose: () => void;
}

export default function RoomModal({ room, onClose }: RoomModalProps) {
  const [selectedStyle, setSelectedStyle] = useState<string | null>(null);
  const [isGenerating, setIsGenerating] = useState(false);
  const [generatedImage, setGeneratedImage] = useState<string | null>(null);

  const generateAfter = trpc.generate.afterImage.useMutation();

  const handleGenerate = async () => {
    if (!selectedStyle || !room.images[0]?.beforeUrl) return;
    setIsGenerating(true);
    try {
      const result = await generateAfter.mutateAsync({
        beforeImageUrl: room.images[0].beforeUrl,
        roomName: room.name,
        style: selectedStyle,
        roomDescription: `${room.name} in a modern apartment`,
      });
      if (result.imageUrl) setGeneratedImage(result.imageUrl);
    } catch (e) {
      console.error(e);
    }
    setIsGenerating(false);
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
      <div className="absolute inset-0 bg-black/70 backdrop-blur-sm" onClick={onClose} />
      <div className="relative w-full max-w-lg max-h-[90vh] overflow-y-auto rounded-2xl border border-border bg-card">
        <div className="flex items-center justify-between p-6 pb-4">
          <div>
            <h2 className="text-xl font-display text-foreground">{room.name}</h2>
            <p className="text-sm text-muted-foreground">{room.images.length} vinkel{room.images.length > 1 ? "e" : ""}</p>
          </div>
          <button onClick={onClose} className="p-2 rounded-lg hover:bg-muted text-muted-foreground"><X size={20} /></button>
        </div>

        {generatedImage && (
          <div className="px-6 mb-4">
            <div className="relative rounded-xl overflow-hidden aspect-video">
              <img src={generatedImage} alt={room.name} className="w-full h-full object-cover" />
            </div>
          </div>
        )}

        <div className="px-6 pb-6">
          <h3 className="text-sm font-medium mb-3 text-muted-foreground uppercase tracking-wider">Vælg stil</h3>
          <div className="space-y-2 mb-6">
            {STYLES.map((style) => (
              <button key={style.id} onClick={() => setSelectedStyle(style.id)} className={`w-full flex items-center gap-4 p-4 rounded-xl border transition-all ${selectedStyle === style.id ? "border-primary bg-primary/10" : "border-border bg-secondary/30 hover:border-primary/30"}`}>
                <div className={`flex items-center justify-center w-10 h-10 rounded-lg ${selectedStyle === style.id ? "bg-primary" : "bg-secondary"}`}><span className="text-lg">{style.icon}</span></div>
                <span className="font-medium text-sm text-foreground">{style.name}</span>
              </button>
            ))}
          </div>
          <Button onClick={handleGenerate} disabled={!selectedStyle || isGenerating} className="w-full h-12 rounded-full bg-primary hover:bg-primary/90 text-primary-foreground">
            {isGenerating ? <><Loader2 size={18} className="mr-2 animate-spin" />Genererer...</> : <><Sparkles size={18} className="mr-2" />Generer {room.name}</>}
          </Button>
        </div>
      </div>
    </div>
  );
}

import { useState } from "react";
import { trpc } from "@/providers/trpc";
import { useNavigate } from "react-router";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Plus, Trash2, Home } from "lucide-react";

const PRESET_ROOMS = [
  { name: "Stue", x: 0, y: 0, width: 4, height: 5, color: "#8B9EB7", description: "Stue med store vinduer" },
  { name: "Køkken", x: 4, y: 0, width: 3, height: 3.5, color: "#D4A373", description: "Køkken med spiseplads" },
  { name: "Soveværelse", x: 0, y: 5, width: 3.5, height: 3.5, color: "#9B8AA5", description: "Master bedroom" },
  { name: "Badeværelse", x: 3.5, y: 5, width: 2, height: 2, color: "#7BA7BC", description: "Badeværelse med brus" },
];

const COLORS = ["#8B9EB7", "#D4A373", "#9B8AA5", "#7BA7BC", "#A8B5A0", "#C9A96E", "#B5C4B1", "#D4A5A5", "#A5B5D4", "#C4B5A5"];

export default function FloorplanForm() {
  const navigate = useNavigate();
  const [name, setName] = useState("");
  const [roomList, setRoomList] = useState(PRESET_ROOMS);

  const createProperty = trpc.property.create.useMutation({
    onSuccess: (data) => { saveRooms(data.id); },
  });
  const setRoomsApi = trpc.property.setRooms.useMutation({
    onSuccess: (_data, variables) => { navigate(`/viewer/${variables.propertyId}`); },
  });

  const saveRooms = (propertyId: number) => {
    setRoomsApi.mutate({
      propertyId,
      rooms: roomList.map((r) => ({ name: r.name, posX: r.x * 10, posY: r.y * 10, width: r.width * 10, height: r.height * 10, color: r.color })),
    });
  };

  const addRoom = () => {
    setRoomList([...roomList, { name: "Nyt rum", x: 0, y: 0, width: 2, height: 2, color: COLORS[roomList.length % COLORS.length], description: "" }]);
  };

  const removeRoom = (index: number) => {
    setRoomList(roomList.filter((_, i) => i !== index));
  };

  const updateRoom = (index: number, field: string, value: string | number) => {
    const updated = [...roomList];
    updated[index] = { ...updated[index], [field]: value };
    setRoomList(updated);
  };

  const handleSubmit = async () => {
    if (!name.trim()) return;
    await createProperty.mutateAsync({ name: name.trim(), floorplanUrl: "" });
  };

  return (
    <div className="w-full">
      <div className="mb-6">
        <Label className="text-sm font-medium text-foreground mb-2 block">Boligens navn</Label>
        <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="f.eks. Villa Sønderskov" className="bg-secondary border-border text-foreground" />
      </div>

      <div className="mb-4 flex items-center justify-between">
        <h3 className="text-sm font-medium text-muted-foreground">Rum ({roomList.length})</h3>
        <Button type="button" variant="outline" size="sm" onClick={addRoom} className="border-border text-muted-foreground hover:text-foreground hover:bg-secondary">
          <Plus size={16} className="mr-1" /> Tilføj rum
        </Button>
      </div>

      <div className="space-y-3 mb-6">
        {roomList.map((room, index) => (
          <div key={index} className="p-4 rounded-lg border border-border bg-secondary/50">
            <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
              <div>
                <Label className="text-xs text-muted-foreground mb-1 block">Navn</Label>
                <Input value={room.name} onChange={(e) => updateRoom(index, "name", e.target.value)} className="h-8 text-sm bg-secondary border-border text-foreground" />
              </div>
              <div>
                <Label className="text-xs text-muted-foreground mb-1 block">Bredde (m)</Label>
                <Input type="number" step="0.5" value={room.width} onChange={(e) => updateRoom(index, "width", parseFloat(e.target.value) || 1)} className="h-8 text-sm bg-secondary border-border text-foreground" />
              </div>
              <div>
                <Label className="text-xs text-muted-foreground mb-1 block">Dybde (m)</Label>
                <Input type="number" step="0.5" value={room.height} onChange={(e) => updateRoom(index, "height", parseFloat(e.target.value) || 1)} className="h-8 text-sm bg-secondary border-border text-foreground" />
              </div>
              <div className="flex items-end gap-2">
                <div className="flex-1">
                  <Label className="text-xs text-muted-foreground mb-1 block">Farve</Label>
                  <div className="flex gap-1">
                    {COLORS.slice(0, 5).map((c) => (
                      <button key={c} onClick={() => updateRoom(index, "color", c)} className="w-5 h-5 rounded-full border-2 transition-transform hover:scale-110" style={{ backgroundColor: c, borderColor: room.color === c ? "hsl(var(--foreground))" : "transparent" }} />
                    ))}
                  </div>
                </div>
                <button onClick={() => removeRoom(index)} className="mb-0.5 p-1.5 rounded hover:bg-muted transition-colors text-muted-foreground"><Trash2 size={14} /></button>
              </div>
            </div>
          </div>
        ))}
      </div>

      <Button onClick={handleSubmit} disabled={!name.trim() || createProperty.isPending} className="w-full h-12 rounded-full bg-primary hover:bg-primary/90 text-primary-foreground font-medium">
        <Home size={18} className="mr-2" />
        {createProperty.isPending ? "Opretter..." : "Opret 3D Visning"}
      </Button>
    </div>
  );
}
