import sharp from "sharp";

export interface CropResult {
  buffer: Buffer;
  metadata: {
    width: number;
    height: number;
    originalBox: [number, number, number, number];
  };
}

/**
 * Crop møbel fra billede med smart padding
 * Sikrer at hele møblet er med + lidt kontekst
 */
export async function cropWithPadding(
  imageBuffer: Buffer,
  box: [number, number, number, number],  // x1, y1, x2, y2 (0-1 normalized)
  paddingPercent: number = 0.12  // 12% padding
): Promise<CropResult> {

  const image = sharp(imageBuffer);
  const metadata = await image.metadata();

  const imgWidth = metadata.width!;
  const imgHeight = metadata.height!;

  // Konverter normalized til pixels
  const x1 = Math.round(box[0] * imgWidth);
  const y1 = Math.round(box[1] * imgHeight);
  const x2 = Math.round(box[2] * imgWidth);
  const y2 = Math.round(box[3] * imgHeight);

  const boxWidth = x2 - x1;
  const boxHeight = y2 - y1;

  // Beregn padding
  const padX = Math.round(boxWidth * paddingPercent);
  const padY = Math.round(boxHeight * paddingPercent);

  // Sikrer at vi ikke går ud over billedet
  const left = Math.max(0, x1 - padX);
  const top = Math.max(0, y1 - padY);
  const right = Math.min(imgWidth, x2 + padX);
  const bottom = Math.min(imgHeight, y2 + padY);

  const cropWidth = right - left;
  const cropHeight = bottom - top;

  // Crop og resize til CLIP input (224x224)
  const croppedBuffer = await image
    .extract({
      left,
      top,
      width: cropWidth,
      height: cropHeight
    })
    .resize(224, 224, {
      fit: 'cover',
      position: 'center'
    })
    .jpeg({ quality: 90 })
    .toBuffer();

  return {
    buffer: croppedBuffer,
    metadata: {
      width: cropWidth,
      height: cropHeight,
      originalBox: box
    }
  };
}

/**
 * Hvis crop fejler, returnér hele billedet (fallback)
 */
export async function safeCrop(
  imageBuffer: Buffer,
  box: [number, number, number, number]
): Promise<Buffer> {
  try {
    const result = await cropWithPadding(imageBuffer, box);
    return result.buffer;
  } catch (e) {
    console.warn("Crop failed, using full image:", e);
    // Resize hele billedet til 224x224
    return sharp(imageBuffer)
      .resize(224, 224, { fit: 'cover' })
      .jpeg({ quality: 90 })
      .toBuffer();
  }
}
