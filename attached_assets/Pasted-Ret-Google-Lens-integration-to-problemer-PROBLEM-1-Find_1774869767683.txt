PROBLEM 1: Finder kun ét produkt
LØSNING: Analysér hele billedet og find ALLE produkter

Ændr i server/google_lens.ts:

export async function analyzeFullImage(imageUrl: string, budget: number) {
  // Del budget op på forventede produkter (estimer 4-8 produkter per rum)
  const estimatedProducts = [
    { type: 'sofa', keywords: 'sofa couch', budgetShare: 0.30 },
    { type: 'bord', keywords: 'table sofabord', budgetShare: 0.15 },
    { type: 'lampe', keywords: 'lamp light gulvlampe', budgetShare: 0.10 },
    { type: 'reol', keywords: 'shelf reol skænk', budgetShare: 0.15 },
    { type: 'tæppe', keywords: 'rug carpet gulvtæppe', budgetShare: 0.10 },
    { type: 'accessories', keywords: 'pillow vase pude', budgetShare: 0.20 }
  ];
  
  const allResults = [];
  
  for (const product of estimatedProducts) {
    // Søg specifikt efter hver produkttype
    const response = await axios.get('https://serpapi.com/search', {
      params: {
        engine: 'google_lens',
        url: imageUrl,
        api_key: process.env.SERPAPI_KEY,
        country: 'dk',
        hl: 'da',
        q: product.keywords  // Specifik søgning per produkttype
      },
      timeout: 10000
    });
    
    // Filtrer KUN danske butikker med DKK
    const danishResults = response.data.visual_matches?.filter((item: any) => {
      const price = item.price;
      if (!price) return false;
      
      // Tjek at valuta er DKK eller EUR (nært nok)
      const isDanishCurrency = price.currency === 'DKK' || price.currency === 'EUR';
      
      // Tjek at butik er dansk/europæisk
      const danishStore = ['IKEA', 'JYSK', 'ILVA', 'Bolia', 'HAY', 'Bahne', 'Menu', 'BoConcept', 'Normann Copenhagen'].some(
        store => item.source?.toLowerCase().includes(store.toLowerCase())
      );
      
      // Tjek at domain er .dk eller europæisk
      const isDanishDomain = item.link?.includes('.dk/') || 
                            item.link?.includes('ikea.com/dk/') ||
                            item.link?.includes('jysk.dk') ||
                            item.link?.includes('ilva.dk');
      
      return isDanishCurrency && (danishStore || isDanishDomain);
    }) || [];
    
    if (danishResults.length > 0) {
      // Tag det bedste match (laveste pris nærmest budgetShare)
      const targetBudget = budget * product.budgetShare;
      const bestMatch = danishResults.sort((a: any, b: any) => {
        const diffA = Math.abs(parseFloat(a.price.value) - targetBudget);
        const diffB = Math.abs(parseFloat(b.price.value) - targetBudget);
        return diffA - diffB;
      })[0];
      
      allResults.push({
        name: product.type,
        source: bestMatch.source,
        price: parseFloat(bestMatch.price.value),
        currency: 'DKK',  // Force DKK
        link: bestMatch.link,
        thumbnail: bestMatch.thumbnail,
        inStock: bestMatch.in_stock,
        targetBudget: Math.round(targetBudget),
        method: 'google_lens'
      });
    }
  }
  
  return allResults;  // Returner ALLE fundne produkter
}

const isDanishResult = (item: any) => {
  // MÅ være DKK
  if (item.price?.currency !== 'DKK') return false;
  
  // MÅ være .dk domæne eller kendt dansk butik
  const link = item.link || '';
  const source = item.source || '';
  
  const danishDomains = [
    '.dk/', 'ikea.com/dk/', 'jysk.dk', 'ilva.dk', 'bolia.com/da-dk/',
    'hay.dk', 'bahne.dk', 'menu.as'
  ];
  
  const isDanishDomain = danishDomains.some(d => link.includes(d));
  
  const danishStores = ['IKEA Danmark', 'JYSK', 'ILVA', 'Bolia', 'HAY', 'Bahne', 'Menu'];
  const isDanishStore = danishStores.some(s => source.includes(s));
  
  return isDanishDomain || isDanishStore;
};

// Brug dette filter i stedet for det gamle

PROBLEM 3: Fallback hvis ingen danske resultater
Hvis Google Lens IKKE finder nok danske produkter, brug OpenAI til at generere søgeord specifikt til danske butikker:

export async function generateDanishSearchTerms(imageUrl: string, roomType: string, style: string) {
  const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
  
  const response = await openai.chat.completions.create({
    model: "gpt-4o",
    messages: [{
      role: "user",
      content: [{
        type: "text",
        text: `Analyser dette ${roomType} i ${style} stil.
        Find ALLE synlige møbler og indretning.
        For hvert produkt, angiv:
        1. Præcis dansk betegnelse (f.eks. "3-pers sofa" ikke bare "sofa")
        2. Materiale og farve
        3. Anslået prisniveau i DKK
        
        Returner JSON:
        {
          "products": [
            {
              "name": "3-pers sofa",
              "searchTerms": "3-pers sofa grå stof dansk design",
              "estimatedPrice": 8000,
              "stores": ["IKEA", "Bolia", "ILVA"]
            }
          ]
        }`
      }, {
        type: "image_url",
        image_url: { url: imageUrl, detail: "high" }
      }]
    }],
    response_format: { type: "json_object" }
  });
  
  return JSON.parse(response.choices[0].message.content);
}
