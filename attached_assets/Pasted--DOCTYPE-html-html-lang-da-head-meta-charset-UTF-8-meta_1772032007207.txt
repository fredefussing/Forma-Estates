    <script>
        // State
        let chatOpen = false;
        let userPreferences = [];
        let conversationHistory = [];

        // Get URL parameters (from quiz)
        const urlParams = new URLSearchParams(window.location.search);
        const room = urlParams.get('room') || 'Stue';
        const style = urlParams.get('style') || 'Skandinavisk';
        const budget = urlParams.get('budget') || '25000';

        // Display selected values
        document.getElementById('selectedRoom').textContent = room;
        document.getElementById('selectedStyle').textContent = style;
        document.getElementById('selectedBudget').textContent = 
            budget === '10000' ? '10.000 kr' : 
            budget === '25000' ? '25.000 kr' : '50.000+ kr';

        function toggleChat() {
            chatOpen = !chatOpen;
            const container = document.getElementById('chatContainer');
            const btn = document.getElementById('toggleChat');
            
            if (chatOpen) {
                container.classList.add('active');
                btn.classList.add('active');
                btn.innerHTML = '<span>✕</span><span>Luk tilpasning</span>';
                document.getElementById('chatInput').focus();
            } else {
                container.classList.remove('active');
                btn.classList.remove('active');
                btn.innerHTML = '<span>💬</span><span>Tilføj specifikke ønsker (valgfrit)</span>';
            }
        }

        function handleKeyPress(event) {
            if (event.key === 'Enter') {
                sendMessage();
            }
        }

        function sendSuggestion(text) {
            document.getElementById('chatInput').value = text;
            sendMessage();
        }

        function sendMessage() {
            const input = document.getElementById('chatInput');
            const message = input.value.trim();
            
            if (!message) return;

            // Add user message
            addMessage(message, 'user');
            userPreferences.push(message);
            conversationHistory.push({ role: 'user', text: message });
            
            // Clear input
            input.value = '';
            
            // Show typing
            document.getElementById('typingIndicator').classList.add('active');
            
            // Simulate bot response
            setTimeout(() => {
                document.getElementById('typingIndicator').classList.remove('active');
                const response = generateResponse(message);
                addMessage(response, 'bot');
                conversationHistory.push({ role: 'bot', text: response });
                updatePreferencesDisplay();
            }, 800);
        }

        function addMessage(text, sender) {
            const container = document.getElementById('chatMessages');
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${sender}`;
            messageDiv.innerHTML = `<div class="message-bubble">${text}</div>`;
            container.appendChild(messageDiv);
            container.scrollTop = container.scrollHeight;
        }

        function generateResponse(userText) {
            const text = userText.toLowerCase();
            
            if (text.includes('sort') || text.includes('mørk')) {
                return "Perfekt! Jeg vil sørge for at inkludere mørke elementer i dit nye design. Det vil skabe en flot kontrast til din stil.";
            }
            else if (text.includes('sofa') || text.includes('stol') || text.includes('møbel')) {
                return "Jeg noterer dit ønske om specifikke møbler. Bemærk at AI'en vil generere et nyt billede med lignende møbler - din endelige indkøbsliste vil indeholde konkrete forslag inden for dit budget.";
            }
            else if (text.includes('lys') || text.includes('lyst') || text.includes('hvid')) {
                return "Godt valg! Jeg vil fokusere på at skabe et lyst og åbent rum. Det passer perfekt til din valgte stil.";
            }
            else if (text.includes('træ') || text.includes('eg') || text.includes('naturlig')) {
                return "Naturlige materialer er altid et hit! Jeg vil inkludere varme trætoner i designet.";
            }
            else if (text.includes('behold') || text.includes('beholde')) {
                return "Forstået! Jeg vil tage højde for at bevare nogle eksisterende elementer. Husk at AI-billedet er inspiration - vi finder de rigtige produkter til dig manuelt.";
            }
            else {
                return "Spændende ønske! Jeg vil inkludere dette i dit design. Har du andre specifikke ønsker, eller er du klar til at se dit nye rum?";
            }
        }

        function updatePreferencesDisplay() {
            const display = document.getElementById('preferencesDisplay');
            const list = document.getElementById('preferencesList');
            
            if (userPreferences.length > 0) {
                display.style.display = 'block';
                list.innerHTML = userPreferences.map(p => 
                    `<span class="preferences-tag">${p}</span>`
                ).join('');
            }
        }

        function skipCustomization() {
            generateDesign();
        }

        function generateDesign() {
            // Build final prompt for Collov
            const promptData = {
                room: room,
                style: style,
                budget: budget,
                preferences: userPreferences,
                conversation: conversationHistory
            };

            // Store for next page
            localStorage.setItem('nordicSketch_preferences', JSON.stringify(promptData));

            // Redirect to generation page
            window.location.href = `/generer?room=${encodeURIComponent(room)}&style=${encodeURIComponent(style)}&budget=${budget}&preferences=${encodeURIComponent(JSON.stringify(userPreferences))}`;
        }

        // Check if preferences exist from before
        window.onload = function() {
            const saved = localStorage.getItem('nordicSketch_preferences');
            if (saved) {
                const data = JSON.parse(saved);
                if (data.preferences && data.preferences.length > 0) {
                    userPreferences = data.preferences;
                    updatePreferencesDisplay();
                }
            }
        };
    </script>
