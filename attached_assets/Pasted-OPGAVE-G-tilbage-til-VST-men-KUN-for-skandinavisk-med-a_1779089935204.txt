2. VST MED AUTOMATISK FALLBACK:

async function runDesignWorkflow(originalUrl, roomType, style) {
  const styleLower = style.toLowerCase();
  const mode = process.env.SOLID10_MODE || 'skandi';
  
  // Bestem om VST skal bruges
  const useVst = (mode === 'none') || 
                 (mode === 'skandi' && styleLower === 'scandinavian');
  
  if (!useVst) {
    console.log('[Workflow] Using SOLID10 for', style);
    return runSolid10Design(originalUrl, roomType, style);
  }
  
  // PRØV VST med 60 sek timeout
  console.log('[Workflow] Trying VST for', style);
  try {
    const vstResult = await Promise.race([
      runVstWorkflow(originalUrl, roomType, style),
      new Promise((_, reject) => 
        setTimeout(() => reject(new Error('VST_TIMEOUT')), 60000)
      )
    ]);
    
    console.log('[Workflow] VST success');
    return vstResult;
    
  } catch (err) {
    // VST fejlede eller timeout → fallback til SOLID10
    console.log('[Workflow] VST failed/timeout, falling back to SOLID10');
    return runSolid10Design(originalUrl, roomType, style);
  }
}
