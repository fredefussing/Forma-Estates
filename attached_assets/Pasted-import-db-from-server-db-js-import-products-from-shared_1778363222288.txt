function guessStyle(name, category, shop) {
  const text = `${name} ${category || ''}`.toLowerCase();
  
  for (const rule of STYLE_RULES) {
    if (rule.pattern.test(text)) return rule.style;
  }
  
  // Fallback: forhandler default
  const shopDefault = SHOP_STYLES[shop?.toLowerCase()];
  if (shopDefault && Math.random() < 0.6) return shopDefault;
  
  return "unknown";
}

function guessColor(name) {
  const text = name.toLowerCase();
  for (const [color, regex] of Object.entries(COLOR_MAP)) {
    if (regex.test(text)) return color;
  }
  return "neutral";
}

function guessMaterial(name) {
  const text = name.toLowerCase();
  for (const [mat, regex] of Object.entries(MATERIAL_MAP)) {
    if (regex.test(text)) return mat;
  }
  return "blandet";
}

function guessType(name) {
  const text = name.toLowerCase();
  for (const [type, regex] of Object.entries(TYPE_MAP)) {
    if (regex.test(text)) return type;
  }
  return "møbel";
}

function guessSize(name, price) {
  if (/xl|stor|large|kæmpe|180|200|220|240/.test(name)) return "stor";
  if (/lille|small|mini|compact|60|80|100/.test(name)) return "lille";
  if (price > 8000) return "stor";
  if (price < 1500) return "lille";
  return "mellem";
}

function guessForm(name) {
  if (/rund|cirkel|oval|bue/.test(name)) return "rund";
  if (/kvadrat|firkant|cube/.test(name)) return "kvadratisk";
  if (/rektangulær|lang|bord|sofa/.test(name)) return "rektangulær";
  return "organisk";
}

async function tagAll() {
  console.log("Starting batch tagging...");
  
  const untagged = await db.query.products.findMany({
    where: sql`${products.tag_processed} = false OR ${products.tag_processed} IS NULL`,
    limit: 250000
  });

  console.log(`Found ${untagged.length} untagged products`);
  
  let processed = 0;
  let batch = [];

  for (const p of untagged) {
    const existingTags = typeof p.tags === 'string' 
      ? JSON.parse(p.tags) 
      : (p.tags || {});
    
    const style = guessStyle(p.name, p.category, p.shop);
    const color = existingTags.color || guessColor(p.name);
    const material = existingTags.material || guessMaterial(p.name);
    const type = existingTags.type || guessType(p.name);
    const size = guessSize(p.name, p.price);
    const form = guessForm(p.name);
    
    const newTags = {
      ...existingTags,
      type,
      subtype: existingTags.subtype || type,
      color,
      color_family: color,
      material,
      material_family: material,
      style,
      size,
      form,
      tag_confidence: style === "unknown" ? 0.4 : 0.75,
      tag_processed: true,
      tag_processed_at: new Date().toISOString()
    };

    batch.push({
      id: p.id,
      tags: JSON.stringify(newTags)
    });

    // Bulk update every 500
    if (batch.length >= 500) {
      await db.update(products)
        .set({
          tags: sql`CASE ${products.id} ${sql.join(
            batch.map(b => sql`WHEN ${b.id} THEN ${b.tags}::jsonb`)
          )} END`,
          tag_processed: true,
          tag_confidence: 0.75
        })
        .where(inArray(products.id, batch.map(b => b.id)));
      
      processed += batch.length;
      batch = [];
      
      if (processed % 5000 === 0) {
        console.log(`Progress: ${processed}/${untagged.length} (${(processed/untagged.length*100).toFixed(1)}%)`);
      }
    }
  }

  // Final batch
  if (batch.length > 0) {
    await db.update(products)
      .set({
        tags: sql`CASE ${products.id} ${sql.join(
          batch.map(b => sql`WHEN ${b.id} THEN ${b.tags}::jsonb`)
        )} END`,
        tag_processed: true
      })
      .where(inArray(products.id, batch.map(b => b.id)));
    
    processed += batch.length;
  }

  console.log(`\n✅ Done! Tagged ${processed} products`);
  
  // Verify
  const total = await db.select({ count: sql`count(*)` }).from(products);
  const tagged = await db.select({ count: sql`count(*)` }).from(products)
    .where(sql`${products.tag_processed} = true`);
  
  console.log(`Total: ${total[0].count}`);
  console.log(`Tagged: ${tagged[0].count} (${(tagged[0].count/total[0].count*100).toFixed(1)}%)`);
}

tagAll()

    console.error("Error:", err);
    process.exit(1);
  })
  .finally(() => process.exit(0));

export async function findProductsForDesign(
  aiImageUrl: string,
  roomType: string,
  styleParam: string,
  budget: string,
  limit: number = 8
) {
  // 1. Beskriv billede
  const description = await describeAiImage(aiImageUrl);
  const parsed = parseDescription(description);
  
  console.log("AI description:", description);
  console.log("Parsed:", parsed);

  // 2. Byg SQL where clause
  const conditions = [sql`${products.tag_processed} = true`];
  
  // Stil: brug parsed OR parameter
  const targetStyle = parsed.styles[0] || styleParam || "scandinavian";
  conditions.push(sql`(${products.tags}->>'style' = ${targetStyle} OR ${products.tags}->>'style' = 'unknown')`);
  
  // Type: vigtigst
  if (parsed.types.length > 0) {
    conditions.push(sql`${products.tags}->>'type' = ANY(${parsed.types})`);
  }
  
  // Farve: medium vigtig
  if (parsed.colors.length > 0) {
    conditions.push(sql`(${products.tags}->>'color' = ANY(${parsed.colors}) OR ${products.tags}->>'color_family' = ANY(${parsed.colors}))`);
  }
  
  // Materiale: nice-to-have
  if (parsed.materials.length > 0) {
    conditions.push(sql`(${products.tags}->>'material' = ANY(${parsed.materials}) OR ${products.tags}->>'material_family' = ANY(${parsed.materials}))`);
  }
  
  // Budget filter
  const priceRanges = {
    budget: [0, 5000],
    standard: [2000, 15000],
    luxury: [8000, 999999]
  };
  const [minPrice, maxPrice] = priceRanges[budget as keyof typeof priceRanges] || [0, 999999];
  conditions.push(sql`${products.price} >= ${minPrice} AND ${products.price} <= ${maxPrice}`);

  // 3. Søg med scoring
  const results = await db.query.products.findMany({
    where: and(...conditions),
    limit: limit * 3, // Hent flere, sorter efter
    orderBy: desc(sql`
      -- Tag match score (0-100)
      (CASE WHEN ${products.tags}->>'type' = ANY(${parsed.types}) THEN 40 ELSE 0 END) +
      (CASE WHEN ${products.tags}->>'color' = ANY(${parsed.colors}) THEN 25 ELSE 0 END) +
      (CASE WHEN ${products.tags}->>'material' = ANY(${parsed.materials}) THEN 15 ELSE 0 END) +
      (CASE WHEN ${products.tags}->>'style' = ${targetStyle} THEN 20 ELSE 0 END) +
      -- Pris proximity
      (100 - ABS(${products.price} - ${(minPrice + maxPrice) / 2}) / ${(maxPrice - minPrice + 1)} * 50)
    `)
  });

  // 4. Returner top med metadata
  return results.slice(0, limit).map(p => ({
    id: p.id,
    name: p.name,
    price: p.price,
    image: p.image_url,
    shop: p.shop,
    link: p.affiliate_link,
    tags: p.tags,
    score: 0 // Beregnes ovenfor
  }));
}

// ─── TRIN 4: Shop This Style ───
export async function getShopThisStyle(
  designId: number,
  aiImageUrl: string,
  roomType: string,
  style: string,
  budget: string
) {
  const products = await findProductsForDesign(aiImageUrl, roomType, style, budget, 8);
  
  // Grupper efter type for variation
  const grouped: Record<string, typeof products> = {};
  for (const p of products) {
    const type = p.tags?.type || 'møbel';
    if (!grouped[type]) grouped[type] = [];
    grouped[type].push(p);
  }
  
  // Vælg 1-2 per type, max 8 total
  const selected = [];
  for (const items of Object.values(grouped)) {
    selected.push(...items.slice(0, 2));
    if (selected.length >= 8) break;
  }
  
  return selected.slice(0, 8);
}
