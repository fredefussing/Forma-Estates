🔧 BACKEND ÆNDRINGER (app.py)
I getCase() endpoint, beregn disse felter og returner dem:
Python
Copy
@app.route("/api/cases/<case_id>", methods=["GET"])
def get_case(case_id):
    ...
    case = dict(row)
    
    # ===== MARKEDSTID TRACKING =====
    case["days_on_market"] = days_on_market(case["market_date"])
    
    # Hent billeder
    cursor.execute("SELECT * FROM images WHERE case_id = ? ORDER BY image_number ASC", (case_id,))
    images = [dict(r) for r in cursor.fetchall()]
    case["images"] = images
    
    # Beregn dage til første billede
    if images and case["market_date"]:
        first_img_date = min(img["created_at"] for img in images)
        market = datetime.fromisoformat(case["market_date"])
        first = datetime.fromisoformat(first_img_date)
        case["days_to_first_image"] = (first - market).days
    else:
        case["days_to_first_image"] = None
    
    # Beregn dage per billede
    for img in images:
        if case["market_date"]:
            market = datetime.fromisoformat(case["market_date"])
            img_date = datetime.fromisoformat(img["created_at"])
            img["days_after_market"] = (img_date - market).days
        else:
            img["days_after_market"] = 0
    
    return jsonify({"success": True, "case": case})
