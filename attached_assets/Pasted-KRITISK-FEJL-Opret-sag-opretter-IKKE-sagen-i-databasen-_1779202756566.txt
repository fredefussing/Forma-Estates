FIX 5: SAG-DETAIL KOMPONENT — Vis data korrekt
═══════════════════════════════════════

Sørg for at din SagDetail komponent modtager og viser data:

interface SagDetailProps {
  case: {
    id: number;
    caseNo?: string;
    address: string;
    notes?: string;
    marketDateISO?: string;
    createdAt: string;
    status: string;
  };
}

function SagDetail({ case: sag }: SagDetailProps) {
  // Beregn dage på markedet
  const marketDate = sag.marketDateISO ? new Date(sag.marketDateISO) : new Date(sag.createdAt);
  const today = new Date();
  const daysOnMarket = Math.floor((today.getTime() - marketDate.getTime()) / (1000 * 60 * 60 * 24));
  
  return (
    <div className="p-6 space-y-6">
      <div className="flex justify-between items-start">
        <div>
          <h1 className="text-2xl font-bold">
            {sag.caseNo || `Sag #${sag.id}`}
          </h1>
          <p className="text-muted-foreground">{sag.address}</p>
        </div>
        <Badge variant="outline">{sag.status}</Badge>
      </div>
      
      <Card>
        <CardHeader>
          <CardTitle>Sagsinfo</CardTitle>
        </CardHeader>
        <CardContent className="space-y-4">
          <div className="flex justify-between">
            <span>Dage på markedet:</span>
            <span className="font-semibold">{daysOnMarket} dage</span>
          </div>
          {sag.notes && (
            <div>
              <span className="text-muted-foreground">Bemærkninger:</span>
              <p>{sag.notes}</p>
            </div>
          )}
        </CardContent>
      </Card>
      
      <div className="flex gap-4">
        <Button>Upload boligbillede</Button>
        <Button variant="outline">Generer potentiale</Button>
      </div>
    </div>
  );
}

app.post("/api/bolig/cases", async (req, res) => {
  try {
    console.log("📥 POST /api/bolig/cases modtaget");
    console.log("📥 Body:", req.body);
    
    const authHeader = req.headers.authorization;
    console.log("📥 Auth header:", authHeader ? "Tilstede" : "MANGLER");
    
    const user = await verifyFirebaseToken(authHeader);
    console.log("📥 Firebase UID:", user.uid);
    
    const dbUser = await storage.getUserByFirebaseUid(user.uid);
    console.log("📥 DB User:", dbUser ? `ID ${dbUser.id}` : "IKKE FUNDET");
    
    if (!dbUser) {
      console.error("❌ Bruger ikke fundet i DB");
      return res.status(401).json({ error: "Unauthorized" });
    }
    
    const { address, caseNo, notes, marketDateISO } = req.body;
    
    const newCase = await storage.createBoligCase({
      userId: dbUser.id,
      address,
      caseNo,
      notes,
      status: "active",
      marketDateISO: marketDateISO || new Date().toISOString().split('T')[0],
    });
    
    console.log("✅ Sag oprettet:", newCase);
    res.status(201).json(newCase);
    
  } catch (error) {
    console.error("❌ Fejl i POST /api/bolig/cases:", error);
    res.status(500).json({ error: "Internal server error" });
  }
});
