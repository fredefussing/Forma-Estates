import { SearchResult } from '../server/vectorSearch';

interface FurnitureGroup {
  label: string;
  detectionIndex: number;
  products: SearchResult[];
}

interface Props {
  groups: FurnitureGroup[];
  isVisible: boolean;
}

export function ShopSection({ groups, isVisible }: Props) {
  if (!isVisible || groups.length === 0) return null;

  return (
    <div className="mt-8 space-y-8">
      {/* Header */}
      <div className="flex items-center gap-3">
        <div className="w-10 h-10 bg-blue-100 rounded-lg flex items-center justify-center">
          <svg className="w-5 h-5 text-blue-600" fill="none" stroke="currentColor" viewBox="0 0 24 24">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} 
                  d="M16 11V7a4 4 0 00-8 0v4M5 9h14l1 12H4L5 9z" />
          </svg>
        </div>
        <div>
          <h2 className="text-xl font-bold text-gray-900">🛍️ SHOP DIT DESIGN</h2>
          <p className="text-sm text-gray-500">
            {groups.reduce((sum, g) => sum + g.products.length, 0)} produkter matcher dit rum
          </p>
        </div>
      </div>

      {/* Grupperet per møbeltype */}
      {groups.map((group, i) => (
        <div key={i} className="bg-white rounded-xl border border-gray-200 overflow-hidden">
          {/* Gruppe header */}
          <div className="px-5 py-4 border-b border-gray-100 bg-gray-50/50">
            <div className="flex items-center justify-between">
              <div className="flex items-center gap-3">
                <span className="w-8 h-8 bg-blue-600 text-white rounded-full 
                               flex items-center justify-center text-sm font-bold">
                  {group.detectionIndex + 1}
                </span>
                <h3 className="font-semibold text-gray-800 capitalize text-lg">
                  {group.label}
                </h3>
              </div>
              <span className="text-sm text-gray-500">
                {group.products.length} fundet
              </span>
            </div>
          </div>

          {/* Produkter */}
          <div className="p-5">
            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
              {/* Best Match */}
              {group.products.filter(p => p.match_type === 'best').map(p => (
                <ProductCard key={p.id} product={p} type="best" />
              ))}

              {/* Similar */}
              {group.products.filter(p => p.match_type === 'similar').slice(0, 2).map((p, idx) => (
                <ProductCard key={`sim-${idx}`} product={p} type="similar" />
              ))}

              {/* Budget */}
              {group.products.filter(p => p.match_type === 'budget').slice(0, 1).map((p, idx) => (
                <ProductCard key={`bud-${idx}`} product={p} type="budget" />
              ))}
            </div>
          </div>
        </div>
      ))}

      {/* Footer CTA */}
      <div className="text-center py-4">
        <p className="text-sm text-gray-500">
          Alle produkter er affiliate-links — du støtter os når du shopper
        </p>
      </div>
    </div>
  );
}

function ProductCard({ 
  product, 
  type 
}: { 
  product: SearchResult; 
  type: 'best' | 'similar' | 'budget';
}) {
  const config = {
    best: {
      badge: 'BEDST MATCH',
      color: 'bg-green-500',
      border: 'border-green-200',
      shadow: 'shadow-green-50'
    },
    similar: {
      badge: 'LIGNENDE',
      color: 'bg-blue-500',
      border: 'border-blue-200',
      shadow: 'shadow-blue-50'
    },
    budget: {
      badge: 'BUDGET',
      color: 'bg-orange-500',
      border: 'border-orange-200',
      shadow: 'shadow-orange-50'
    }
  };

  const c = config[type];

  return (
    <a
      href={product.affiliate_link}
      target="_blank"
      rel="noopener noreferrer"
      className={`group block rounded-lg border ${c.border} overflow-hidden 
                hover:shadow-lg transition-all duration-200 ${c.shadow}`}
    >
      <div className="relative aspect-[4/3] bg-gray-100">
        <img 
          src={product.image_url} 
          alt={product.name}
          className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
          loading="lazy"
        />
        <span className={`absolute top-3 left-3 ${c.color} text-white text-xs 
                        px-3 py-1.5 rounded-full font-semibold tracking-wide`}>
          {c.badge}
        </span>

        {/* Similarity score */}
        <div className="absolute top-3 right-3 bg-black/60 text-white text-xs 
                      px-2 py-1 rounded backdrop-blur-sm">
          {(product.similarity * 100).toFixed(0)}% match
        </div>
      </div>

      <div className="p-4">
        <p className="text-sm font-medium text-gray-900 line-clamp-2 
                    group-hover:text-blue-600 transition-colors leading-snug">
          {product.name}
        </p>

        {product.price && (
          <div className="mt-2 flex items-baseline gap-1">
            <span className="text-lg font-bold text-gray-900">
              {product.price.toLocaleString('da-DK')} kr
            </span>
            {type === 'budget' && (
              <span className="text-xs text-green-600 font-medium">
                Spar penge
              </span>
            )}
          </div>
        )}

        <div className="mt-3 flex items-center gap-1 text-sm text-blue-600 font-medium 
                      group-hover:gap-2 transition-all">
          <span>Se produkt</span>
          <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} 
                  d="M9 5l7 7-7 7" />
          </svg>
        </div>
      </div>
    </a>
  );
}
