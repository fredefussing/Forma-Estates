🎬 INTERAKTIVITET (JavaScript)
JavaScript
Copy
// Vis/skjul sektioner
function showSection(section) {
  // Gem alle sektioner
  document.querySelectorAll('.page-section').forEach(s => s.style.display = 'none');
  // Vis den valgte
  const target = document.getElementById(section);
  if (target) target.style.display = 'block';
  // Opdater aktiv menu
  document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
  event.target.closest('.nav-item').classList.add('active');
}

// Modal
function openModal(modalId) {
  document.getElementById(modalId + 'Modal').classList.add('active');
}

function closeModal(modalId) {
  document.getElementById(modalId + 'Modal').classList.remove('active');
}

// Opret sag
function createSag() {
  // 1. Validér formular
  // 2. Send til backend
  // 3. Vis succes-besked
  // 4. Luk modal
  // 5. Genindlæs sager
  closeModal('newSag');
  showToast('Sag oprettet!');
}

// Upload og generer
function uploadAndGenerate() {
  // 1. Vis loading
  // 2. Upload billede til /api/generate
  // 3. Vis resultat
  // 4. Gem i sagsmappe
  closeModal('uploadBillede');
  showToast('Billede genereres...');
}

// Toast notification
function showToast(message) {
  const toast = document.createElement('div');
  toast.className = 'toast';
  toast.textContent = message;
  document.body.appendChild(toast);
  setTimeout(() => toast.remove(), 3000);
}
