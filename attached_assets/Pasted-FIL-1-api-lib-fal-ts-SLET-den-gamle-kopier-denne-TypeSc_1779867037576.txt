export async function generate3DFloorplan(prompt: string, aspectRatio: string = "16:9"): Promise<{ imageUrl: string }> {
  const result = await fal.subscribe("fal-ai/nano-banana-2", {
    input: {
      prompt: `Isometric 3D architectural floor plan visualization, ${prompt}, viewed from a 35-degree elevated angle, natural daylight, soft shadows, architectural cutaway style showing interior details, photorealistic 3D render, 8k`,
      aspect_ratio: aspectRatio as any,
      resolution: "2K",
    },
  });
  const imageUrl = (result.data as any).images?.[0]?.url;
  if (!imageUrl) throw new Error("No image generated");
  return { imageUrl };
}

export async function generateAfterImage(beforeImageUrl: string, roomName: string, style: string, roomDescription: string): Promise<{ imageUrl: string }> {
  const stylePrompts: Record<string, string> = {
    moderne: "modern minimalist interior, clean lines, neutral palette, floor-to-ceiling windows, polished floors, warm ambient lighting",
    klassisk: "classic luxury interior, ornate moldings, dark wood paneling, crystal chandelier, rich fabrics, marble fireplace",
    nordisk: "Scandinavian interior, light oak wood, white walls, cozy textiles, natural light, hygge atmosphere, green plants",
    kyst: "coastal interior, ocean views, white and beige palette, natural materials, sheer curtains, indoor palm plants",
    luksus: "luxury penthouse interior, marble floors, gold accents, designer furniture, panoramic windows, dramatic lighting",
  };
  const styleDesc = stylePrompts[style] || stylePrompts.nordisk;
  const result = await fal.subscribe("fal-ai/nano-banana-2", {
    input: {
      prompt: `Renovated ${roomName}, ${styleDesc}. ${roomDescription} Photorealistic architectural visualization, 8k, ultra-detailed, same camera angle and room layout`,
      image_url: beforeImageUrl,
      aspect_ratio: "16:9",
      resolution: "2K",
    } as any,
  });
  const imageUrl = (result.data as any).images?.[0]?.url;
  if (!imageUrl) throw new Error("No after image generated");
  return { imageUrl };
}

export async function generate360Panorama(roomName: string, style: string, roomDescription: string, angle: string): Promise<{ imageUrl: string }> {
  const angleDesc: Record<string, string> = {
    entrance: "viewed from the entrance doorway looking inward",
    window: "viewed from near the large window, natural daylight streaming in",
    corner: "viewed from a corner position showing the full room layout",
    center: "viewed from the center of the room, 360 perspective",
  };
  const result = await fal.subscribe("fal-ai/nano-banana-2", {
    input: {
      prompt: `360 degree equirectangular panorama of a ${roomName}, ${style} interior design, ${angleDesc[angle] || angleDesc.center}. ${roomDescription} Seamless wraparound, complete room visible in all directions, photorealistic, 8k`,
      aspect_ratio: "16:9",
      resolution: "2K",
    },
  });
  const imageUrl = (result.data as any).images?.[0]?.url;
  if (!imageUrl) throw new Error("No panorama generated");
  return { imageUrl };
}

export async function generateAnimationVideo(beforeImageUrl: string, _afterImageUrl: string, roomType: string = "room"): Promise<{ videoUrl: string }> {
  const result = await fal.subscribe("fal-ai/kling-video/v3/pro/image-to-video", {
    input: {
      image_url: beforeImageUrl,
      prompt: `Cinematic renovation of a ${roomType}. Camera pushes forward as the room transforms — walls change, flooring spreads, furniture materializes, warm lighting illuminates. Smooth, elegant, no abrupt cuts. Photorealistic, 8k.`,
      end_image_url: _afterImageUrl,
      duration: "8" as any,
      aspect_ratio: "16:9",
    } as any,
  });
  const videoUrl = (result.data as any).video?.url;
  if (!videoUrl) throw new Error("No video generated");
  return { videoUrl };
}

import { z } from "zod";
import { createRouter, publicQuery } from "../middleware";
import { getDb } from "../queries/connection";
import { renovations } from "@db/schema";
import { eq } from "drizzle-orm";
import { generateAnimationVideo } from "../lib/fal";

export const renovationRouter = createRouter({
  create: publicQuery
    .input(z.object({ name: z.string().min(1), beforeImageUrl: z.string().url(), afterImageUrl: z.string().url(), roomType: z.string().default("room") }))
    .mutation(async ({ input }) => {
      const db = getDb();
      const [renovation] = await db.insert(renovations).values({
        name: input.name,
        beforeImageUrl: input.beforeImageUrl,
        afterImageUrl: input.afterImageUrl,
        roomType: input.roomType,
        status: "pending",
      }).$returningId();
      return { id: renovation.id };
    }),

  generateVideo: publicQuery
    .input(z.object({ id: z.number() }))
    .mutation(async ({ input }) => {
      const db = getDb();
      const [renovation] = await db.select().from(renovations).where(eq(renovations.id, input.id));
      if (!renovation) return { success: false, error: "Not found" };

      await db.update(renovations).set({ status: "generating" }).where(eq(renovations.id, input.id));

      try {
        const result = await generateAnimationVideo(renovation.beforeImageUrl, renovation.afterImageUrl, renovation.roomType || "room");
        await db.update(renovations).set({ videoUrl: result.videoUrl, status: "completed" }).where(eq(renovations.id, input.id));
        return { success: true, videoUrl: result.videoUrl };
      } catch (error: any) {
        await db.update(renovations).set({ status: "failed" }).where(eq(renovations.id, input.id));
        return { success: false, error: error.message };
      }
    }),

  list: publicQuery.query(async () => {
    const db = getDb();
    return db.select().from(renovations).orderBy(renovations.createdAt);
  }),

  get: publicQuery
    .input(z.object({ id: z.number() }))
    .query(async ({ input }) => {
      const db = getDb();
      const [renovation] = await db.select().from(renovations).where(eq(renovations.id, input.id));
      return renovation || null;
    }),

  delete: publicQuery
    .input(z.object({ id: z.number() }))
    .mutation(async ({ input }) => {
      const db = getDb();
      await db.delete(renovations).where(eq(renovations.id, input.id));
      return { success: true };
    }),
});

import { z } from "zod";
import { createRouter, publicQuery } from "../middleware";
import { getDb } from "../queries/connection";
import { properties, propertyRooms, roomImages } from "@db/schema";
import { eq } from "drizzle-orm";

export const propertyRouter = createRouter({
  create: publicQuery
    .input(z.object({ name: z.string().min(1), floorplanUrl: z.string() }))
    .mutation(async ({ input }) => {
      const db = getDb();
      const [prop] = await db.insert(properties).values({
        name: input.name,
        floorplanUrl: input.floorplanUrl,
        status: "uploading",
      }).$returningId();
      return { id: prop.id };
    }),

  setRooms: publicQuery
    .input(z.object({
      propertyId: z.number(),
      rooms: z.array(z.object({ name: z.string(), posX: z.number(), posY: z.number(), width: z.number(), height: z.number(), color: z.string() })),
    }))
    .mutation(async ({ input }) => {
      const db = getDb();
      await db.delete(propertyRooms).where(eq(propertyRooms.propertyId, input.propertyId));
      if (input.rooms.length > 0) {
        await db.insert(propertyRooms).values(input.rooms.map((r) => ({
          propertyId: input.propertyId,
          name: r.name,
          posX: r.posX,
          posY: r.posY,
          width: r.width,
          height: r.height,
          color: r.color,
        })));
      }
      await db.update(properties).set({ status: "mapping" }).where(eq(properties.id, input.propertyId));
      return { success: true };
    }),

  setStyle: publicQuery
    .input(z.object({ propertyId: z.number(), style: z.string() }))
    .mutation(async ({ input }) => {
      const db = getDb();
      await db.update(properties).set({ style: input.style }).where(eq(properties.id, input.propertyId));
      return { success: true };
    }),

  setStatus: publicQuery
    .input(z.object({ propertyId: z.number(), status: z.string() }))
    .mutation(async ({ input }) => {
      const db = getDb();
      await db.update(properties).set({ status: input.status }).where(eq(properties.id, input.propertyId));
      return { success: true };
    }),

  get: publicQuery
    .input(z.object({ id: z.number() }))
    .query(async ({ input }) => {
      const db = getDb();
      const [prop] = await db.select().from(properties).where(eq(properties.id, input.id));
      if (!prop) return null;
      const rooms = await db.select().from(propertyRooms).where(eq(propertyRooms.propertyId, input.id));
      return { ...prop, rooms };
    }),

  list: publicQuery.query(async () => {
    const db = getDb();
    return db.select().from(properties).orderBy(properties.createdAt);
  }),

  delete: publicQuery
    .input(z.object({ id: z.number() }))
    .mutation(async ({ input }) => {
      const db = getDb();
      await db.delete(propertyRooms).where(eq(propertyRooms.propertyId, input.id));
      await db.delete(properties).where(eq(properties.id, input.id));
      return { success: true };
    }),
});

import { useState, useRef, useEffect } from "react";
import { useParams, useNavigate } from "react-router";
import { trpc } from "@/providers/trpc";
import { ArrowLeft, ChevronLeft, ChevronRight, Box } from "lucide-react";

interface Room {
  id: number;
  name: string;
  posX: number;
  posY: number;
  width: number;
  height: number;
  color: string;
  hasImages: number;
  images: RoomImage[];
}

interface RoomImage {
  id: number;
  angle: string;
  beforeUrl: string | null;
  afterUrl: string | null;
  status: string;
}

const ANGLE_LABELS: Record<string, string> = {
  entrance: "Fra indgangen",
  window: "Fra vinduet",
  corner: "Fra hjørnet",
  center: "Fra midten",
};

export default function PropertyViewer() {
  const { id } = useParams<{ id: string }>();
  const navigate = useNavigate();
  const propertyId = parseInt(id || "0");

  const { data: property, isLoading } = trpc.property.get.useQuery(
    { id: propertyId },
    { enabled: propertyId > 0 }
  );

  const [mode, setMode] = useState<"floorplan" | "room">("floorplan");
  const [selectedRoom, setSelectedRoom] = useState<Room | null>(null);
  const [selectedImageIndex, setSelectedImageIndex] = useState(0);
  const [revealPercent, setRevealPercent] = useState(50);
  const [isDragging, setIsDragging] = useState(false);

  const containerRef = useRef<HTMLDivElement>(null);

  if (isLoading) {
    return <div className="min-h-screen flex items-center justify-center bg-background"><Box size={40} className="animate-pulse text-primary" /></div>;
  }

  if (!property) {
    return <div className="min-h-screen flex items-center justify-center bg-background"><p className="text-muted-foreground">Boligen blev ikke fundet</p></div>;
  }

  const rooms = (property.rooms || []) as Room[];
  const currentRoom = selectedRoom || rooms[0] || null;
  const currentImage = currentRoom?.images?.[selectedImageIndex];

  const handleRoomClick = (room: Room) => {
    if (room.images.length === 0) return;
    setSelectedRoom(room);
    setSelectedImageIndex(0);
    setRevealPercent(50);
    setMode("room");
  };

  const handlePointerDown = (clientX: number) => {
    if (!containerRef.current) return;
    const rect = containerRef.current.getBoundingClientRect();
    const x = clientX - rect.left;
    setRevealPercent(Math.max(2, Math.min(98, (x / rect.width) * 100)));
    setIsDragging(true);
  };

  const handlePointerMove = (clientX: number) => {
    if (!isDragging || !containerRef.current) return;
    const rect = containerRef.current.getBoundingClientRect();
    setRevealPercent(Math.max(2, Math.min(98, ((clientX - rect.left) / rect.width) * 100)));
  };

  const handlePointerUp = () => setIsDragging(false);

  useEffect(() => {
    if (!isDragging) return;
    const onMove = (e: MouseEvent) => handlePointerMove(e.clientX);
    const onUp = () => handlePointerUp();
    window.addEventListener("mousemove", onMove);
    window.addEventListener("mouseup", onUp);
    return () => { window.removeEventListener("mousemove", onMove); window.removeEventListener("mouseup", onUp); };
  }, [isDragging]);

  const handleBack = () => { setMode("floorplan"); setSelectedRoom(null); };

  return (
    <div className="min-h-screen bg-background">
      <header className="fixed top-0 left-0 right-0 z-50 bg-background/90 backdrop-blur-xl border-b border-border">
        <div className="max-w-7xl mx-auto h-14 flex items-center justify-between px-4">
          <div className="flex items-center gap-3">
            <button onClick={() => (mode === "room" ? handleBack() : navigate("/"))} className="p-2 rounded-lg hover:bg-muted text-muted-foreground"><ArrowLeft size={18} /></button>
            <h1 className="text-sm font-medium text-foreground">{property.name}</h1>
          </div>
