import { db } from "./db";
import { describeWithVision, getBasicDescription, DANISH_SYNONYMS } from "./describeWithVision";
import { generateClipVector } from "./clip";

export interface SearchResult {
  id: string;
  name: string;
  image_url: string;
  affiliate_link: string;
  price: number | null;
  similarity: number;
  match_type: 'best' | 'similar' | 'budget';
}

export interface FurnitureMatch {
  label: string;
  box: [number, number, number, number];
  products: SearchResult[];
}

/**
 * KERNE-FIX: SQL-first, CLIP-second
 * 1. Vision analyserer møblet
 * 2. SQL filtrerer 243k → 50-200 relevante produkter
 * 3. CLIP finder bedste match BLANDT de filtrerede
 */
export async function findSimilarWithVision(
  cropBuffer: Buffer,
  yoloLabel: string,
  topK: number = 3
): Promise<SearchResult[]> {

  // STEP 1: Vision analyserer croppet (vores "søgemaskine")
  let vision;
  try {
    vision = await describeWithVision(cropBuffer);
    console.log("Vision analysis:", vision);
  } catch (e) {
    console.warn("Vision failed, using YOLO fallback:", e);
    vision = { ...getBasicDescription(yoloLabel), confidence: 0.3 } as any;
  }

  // STEP 2: BYG SQL FILTER
  // Dette reducerer 243k til 50-200 produkter der faktisk matcher
  const { sql, params } = buildSearchSQL(vision, yoloLabel);

  console.log("SQL Filter query:", sql.substring(0, 200) + "...");

  const candidates = await db.query(sql, params);
  console.log(`SQL filter found ${candidates.length} candidates`);

  if (candidates.length === 0) {
    console.warn("No products matched vision filters:", vision);
    // Fallback: søg bredere uden farve/materiale filter
    return fallbackSearch(cropBuffer, yoloLabel, topK);
  }

  // STEP 3: CLIP vector søgning KUN på de filtrerede kandidater
  const cropVector = await generateClipVector(cropBuffer);

  const candidateIds = candidates.map((c: any) => c.id);

  const vectorSQL = `
    SELECT 
      p.id,
      p.name,
      p.image_url,
      p.affiliate_link,
      p.price,
      1 - (p.vector_clip <=> $1::vector) as similarity
    FROM products p
    WHERE p.id = ANY($2)
    ORDER BY p.vector_clip <=> $1::vector
    LIMIT $3
  `;

  const results = await db.query(vectorSQL, [
    cropVector,
    candidateIds,
    Math.min(topK * 3, 20)  // Hent flere så vi kan gruppere varianter
  ]);

  // STEP 4: Post-processing — fjern duplikater og gruppér
  const processed = postProcessResults(results, vision, topK);

  return processed;
}

/**
 * Bygger SQL query med danske synonymer
 * Dette er MAGIEN — vi søger ikke i hele databasen
 */
function buildSearchSQL(vision: any, yoloLabel: string) {
  const conditions: string[] = [];
  const params: any[] = [];
  let paramIdx = 1;

  // TYPE filter — brug både Vision type og YOLO label
  const typeTerms = getTypeTerms(vision.type, yoloLabel);
  if (typeTerms.length > 0) {
    const typeCondition = typeTerms.map((t: string) => 
      `(p.name ILIKE $${paramIdx++} OR p.category ILIKE $${paramIdx++})`
    ).join(' OR ');
    conditions.push(`(${typeCondition})`);
    typeTerms.forEach((t: string) => {
      params.push(`%${t}%`);
      params.push(`%${t}%`);
    });
  }

  // COLOR filter — kritisk for at undgå "beige → brun" fejl
  if (vision.color && vision.color !== 'unknown') {
    const colorTerms = DANISH_SYNONYMS[vision.color] || [vision.color];
    const colorCondition = colorTerms.map((c: string) => 
      `p.name ILIKE $${paramIdx++}`
    ).join(' OR ');
    conditions.push(`(${colorCondition})`);
    colorTerms.forEach((c: string) => params.push(`%${c}%`));
  }

  // MATERIAL filter — kritisk for at undgå "vævet → læder" fejl
  if (vision.material && vision.material !== 'unknown') {
    const matTerms = DANISH_SYNONYMS[vision.material] || [vision.material];
    const matCondition = matTerms.map((m: string) => 
      `p.name ILIKE $${paramIdx++}`
    ).join(' OR ');
    conditions.push(`(${matCondition})`);
    matTerms.forEach((m: string) => params.push(`%${m}%`));
  }

  // STYLE filter — hjælper med skandinavisk matching
  if (vision.style && vision.style !== 'unknown') {
    const styleTerms = DANISH_SYNONYMS[vision.style] || [vision.style];
    const styleCondition = styleTerms.map((s: string) => 
      `p.name ILIKE $${paramIdx++} OR p.category ILIKE $${paramIdx++}`
    ).join(' OR ');
    conditions.push(`(${styleCondition})`);
    styleTerms.forEach((s: string) => {
      params.push(`%${s}%`);
      params.push(`%${s}%`);
    });
  }

  // INDOOR/OUTDOOR filter — undgå at indoor møbler matcher udendørs
  if (vision.indoor) {
    conditions.push(`(
      p.name NOT ILIKE $${paramIdx++} 
      AND p.name NOT ILIKE $${paramIdx++} 
      AND p.name NOT ILIKE $${paramIdx++}
      AND p.name NOT ILIKE $${paramIdx++}
      AND p.category NOT ILIKE $${paramIdx++}
    )`);
    params.push('%udendørs%');
    params.push('%have%');
    params.push('%terrasse%');
    params.push('%solstol%');
    params.push('%udendørs%');
  }

  // Pris range — ekskluder ekstremt dyre/ billige hvis vi har prisdata
  // (valgfrit — kan fjernes hvis pris ikke er vigtig)

  // Sammenbyg SQL
  const whereClause = conditions.join(' AND ');

  const sql = `
    SELECT 
      p.id,
      p.name,
      p.image_url,
      p.affiliate_link,
      p.price,
      p.vector_clip
    FROM products p
    WHERE ${whereClause}
    ORDER BY p.price ASC NULLS LAST
    LIMIT 200
  `;

  return { sql, params };
}

/**
 * Hjælpefunktion: få alle type-termer (engelsk + dansk)
 */
function getTypeTerms(visionType: string, yoloLabel: string): string[] {
  const terms = new Set<string>();

  // Vision type
  if (visionType && visionType !== 'unknown') {
    const visionTerms = DANISH_SYNONYMS[visionType] || [visionType];
    visionTerms.forEach((t: string) => terms.add(t));
  }

  // YOLO label fallback
  const yoloMap: Record<string, string[]> = {
    'chair': ['stol', 'lænestol', 'spisestol'],
    'couch': ['sofa', 'sovesofa'],
    'bed': ['seng', 'sengestel'],
    'dining table': ['spisebord', 'bord'],
    'potted plant': ['plante', 'potte'],
    'vase': ['vase'],
    'book': ['bog', 'reol'],
    'clock': ['ur', 'vægur']
  };

  const yoloTerms = yoloMap[yoloLabel] || [yoloLabel];
  yoloTerms.forEach((t: string) => terms.add(t));

  return Array.from(terms);
}

/**
 * Post-process: fjern duplikater (varianter), gruppér, vælg Best/Similar/Budget
 */
function postProcessResults(
  results: any[],
  vision: any,
  topK: number
): SearchResult[] {

  // Fjern duplikater baseret på produktnavn (uden størrelse/farve variant)
  const seen = new Set<string>();
  const unique: any[] = [];

  for (const r of results) {
    // Normaliser navn: fjern størrelser, farver, materialer for at finde "base product"
    const baseName = r.name
      .replace(/\d+\s*x\s*\d+/gi, '')  // fjern "140 x 200"
      .replace(/\d+\s*cm/gi, '')       // fjern "45cm"
      .replace(/-\s*\w+\s*\/\s*/g, '') // fjern "- Grå /"
      .trim();

    if (!seen.has(baseName)) {
      seen.add(baseName);
      unique.push(r);
    }
  }

  // Sorter efter similarity
  unique.sort((a, b) => b.similarity - a.similarity);

  // Vælg Best, Similar, Budget
  const best = unique[0];
  const similar = unique.slice(1, 3);

  // Budget = billigere end best, stadig god similarity
  const budget = unique
    .filter((u: any) => u.price && best.price && u.price < best.price * 0.8)
    .slice(0, 2);

  const output: SearchResult[] = [];

  if (best) output.push({ ...best, match_type: 'best' });
  similar.forEach((s: any) => output.push({ ...s, match_type: 'similar' }));
  budget.forEach((b: any) => output.push({ ...b, match_type: 'budget' }));

  return output.slice(0, topK);
}

/**
 * Fallback: hvis Vision + SQL finder 0 produkter
 * Brug basic CLIP søgning uden filtre
 */
async function fallbackSearch(
  cropBuffer: Buffer,
  yoloLabel: string,
  topK: number
): Promise<SearchResult[]> {
  console.log("Using fallback search...");

  const cropVector = await generateClipVector(cropBuffer);

  const sql = `
    SELECT 
      p.id,
      p.name,
      p.image_url,
      p.affiliate_link,
      p.price,
      1 - (p.vector_clip <=> $1::vector) as similarity
    FROM products p
    WHERE p.category ILIKE $2 OR p.name ILIKE $2
    ORDER BY p.vector_clip <=> $1::vector
    LIMIT $3
  `;

  const results = await db.query(sql, [cropVector, `%${yoloLabel}%`, topK]);

  return results.map((r: any) => ({ ...r, match_type: 'best' as const }));
}
