def allowed_file(filename):
    return "." in filename and filename.rsplit(".", 1)[1].lower() in ALLOWED_EXTENSIONS


def save_base64_image(base64_string, folder):
    """Gem base64 billede til fil og returner sti"""
    try:
        # Fjern data:image/png;base64, prefix hvis det findes
        if "," in base64_string:
            base64_string = base64_string.split(",")[1]
        
        image_data = base64.b64decode(base64_string)
        filename = f"{uuid.uuid4().hex}.png"
        filepath = os.path.join(folder, filename)
        
        with open(filepath, "wb") as f:
            f.write(image_data)
        
        return filepath
    except Exception as e:
        print(f"Fejl ved gem af billede: {e}")
        return None


def image_to_base64(image_path):
    """Konverter billede til base64 string"""
    try:
        with open(image_path, "rb") as f:
            return base64.b64encode(f.read()).decode("utf-8")
    except Exception as e:
        print(f"Fejl ved konvertering til base64: {e}")
        return None

@app.route("/api/generate", methods=["POST"])
def generate():
    """
    HOVEDENDPOINT: Generer AI-visualisering
    =======================================
    
    Modtager:
        - image: Billede fil (multipart/form-data) ELLER base64 string (JSON)
        - style: Stilart (modern, scandinavian, classic, luxury, minimal, nordic)
        - room: Rumtype (kitchen, livingroom, bathroom, bedroom, office, dining)
    
    Returnerer:
        - success: true/false
        - image_url: Base64 encoded resultat billede
        - processing_time: Tid brugt på generering
    
    Eksempel curl:
        curl -X POST https://DIN-REPLIT.replit.app/api/generate \
          -F "image=@dit_billede.jpg" \
          -F "style=scandinavian" \
          -F "room=livingroom"
    """
    start_time = time.time()
    
    try:
        # ---------------------------------------------------------------------
        # 1. Modtag billede
        # ---------------------------------------------------------------------
        input_path = None
        
        # Mulighed A: Billede som fil upload (multipart/form-data)
        if "image" in request.files:
            file = request.files["image"]
            if file.filename == "":
                return jsonify({"success": False, "error": "Ingen fil valgt"}), 400
            
            if file and allowed_file(file.filename):
                filename = secure_filename(f"{uuid.uuid4().hex}_{file.filename}")
                input_path = os.path.join(UPLOAD_FOLDER, filename)
                file.save(input_path)
        
        # Mulighed B: Billede som base64 string (JSON)
        elif request.is_json and "image_base64" in request.json:
            base64_str = request.json["image_base64"]
            input_path = save_base64_image(base64_str, UPLOAD_FOLDER)
        
        if not input_path:
            return jsonify({
                "success": False, 
                "error": "Intet billede modtaget. Send som fil (image) eller base64 (image_base64)"
            }), 400
        
        # ---------------------------------------------------------------------
        # 2. Modtag parametre
        # ---------------------------------------------------------------------
        style = request.form.get("style", request.json.get("style", "modern") if request.is_json else "modern")
        room_type = request.form.get("room", request.json.get("room", "livingroom") if request.is_json else "livingroom")
        
        # Validér style
        valid_styles = ["modern", "scandinavian", "classic", "luxury", "minimal", "nordic", "industrial", "bohemian"]
        if style not in valid_styles:
            style = "modern"  # Default
        
        # Validér room
        valid_rooms = ["kitchen", "livingroom", "bathroom", "bedroom", "office", "dining", "entrance"]
        if room_type not in valid_rooms:
            room_type = "livingroom"  # Default
        
        print(f" Genererer: style={style}, room={room_type}")
        
        # ---------------------------------------------------------------------
        # 3. Kør AI-generering
        # ---------------------------------------------------------------------
        result_path = generate_ai_image(input_path, style, room_type)
        
        if not result_path or not os.path.exists(result_path):
            return jsonify({"success": False, "error": "Generering fejlede"}), 500
        
        # ---------------------------------------------------------------------
        # 4. Returner resultat som base64
        # ---------------------------------------------------------------------
        result_base64 = image_to_base64(result_path)
        
        if not result_base64:
            return jsonify({"success": False, "error": "Kunne ikke læse resultat"}), 500
        
        processing_time = round(time.time() - start_time, 2)
        
        return jsonify({
            "success": True,
            "image_base64": f"data:image/png;base64,{result_base64}",
            "style": style,
            "room": room_type,
            "processing_time": processing_time
        })
        
    except Exception as e:
        print(f" Fejl: {str(e)}")
        return jsonify({"success": False, "error": str(e)}), 500

async function generateImage(imageFile: File, style: string, room: string) {
  const formData = new FormData();
  formData.append("image", imageFile);
  formData.append("style", style);
  formData.append("room", room);
  
  const response = await fetch(`${API_URL}/api/generate`, {
    method: "POST",
    body: formData,
  });
  
  const data = await response.json();
  if (data.success) {
    return data.image_base64; // "data:image/png;base64,..."
  }
  throw new Error(data.error);
