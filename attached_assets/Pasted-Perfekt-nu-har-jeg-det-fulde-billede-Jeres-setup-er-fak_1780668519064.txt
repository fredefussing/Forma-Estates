// showcase.ts — tilføj dette før assemble-trinnet
function getClipDuration(roomType: string, beatInterval: number): number {
  const base = beatInterval; // typisk 2-3 sekunder per beat
  switch (roomType) {
    case 'udsigt': case 'altan': case 'terrasse': return base * 1.8;  // 5-6 sek — imponerende
    case 'køkken': case 'stue': return base * 1.4;                    // 4 sek — vigtige rum
    case 'soveværelse': case 'badeværelse': return base * 1.0;        // 3 sek — standard
    case 'entre': case 'gang': case 'bryggers': return base * 0.6;    // 1.5-2 sek — hurtigt
    default: return base;
  }
}

// [0:v][1:v]xfade=transition=fade:duration=1.0:offset=4.0[f1];
// [f1][2:v]xfade=transition=fade:duration=1.0:offset=7.5[f2]

function buildCrossfadeFilter(
  clipCount: number, 
  clipDurations: number[],  // fx [5, 4, 2, 4.5, ...]
  fadeDuration: number = 1.0
): string {
  let filter = '';
  let offset = 0;
  
  for (let i = 0; i < clipCount - 1; i++) {
    const inputA = i === 0 ? '[0:v]' : `[f${i}]`;
    const inputB = `[${i + 1}:v]`;
    const output = i === clipCount - 2 ? '[outv]' : `[f${i + 1}]`;
    
    offset += clipDurations[i] - (i > 0 ? fadeDuration : 0);
    filter += `${inputA}${inputB}xfade=transition=fade:duration=${fadeDuration}:offset=${offset.toFixed(2)}${output}`;
    if (i < clipCount - 2) filter += ';';
n  }
  return filter;
}
3. Beat-sync med varierende klip-længder (den smarte del)
I stedet for at alle klip skifter på beat, skal kun nogle klip skifte på beat — resten følger impact-varigheden. Det giver en dynamisk rytme:
TypeScript
// Beat detection (brug librosa eller aubio)
const beatTimes = await detectBeats(audioPath); // [0, 2.4, 4.8, 7.2, 9.6, ...]

// Byg "rhythm map" — hvilke klip skifter på beat?
function buildRhythmMap(
  clipCount: number, 
  impactScores: number[]
): boolean[] {
  // Første og sidste klip + højeste impact = beat-sync
  // Resten = fri varighed
  const sorted = [...impactScores].map((s, i) => ({ s, i }))
    .sort((a, b) => b.s - a.s);
  
  const beatSyncIndices = new Set([0, clipCount - 1]); // altid start+slut
  sorted.slice(0, Math.ceil(clipCount * 0.4)).forEach(x => beatSyncIndices.add(x.i));
  
  return Array.from({ length: clipCount }, (_, i) => beatSyncIndices.has(i));
}
Resultat: Videoen føles synkroniseret til musikken uden at være mekanisk. Nogle klip er korte og snappy, andre strækker sig over flere beats og giver plads til at nyde rummet.
