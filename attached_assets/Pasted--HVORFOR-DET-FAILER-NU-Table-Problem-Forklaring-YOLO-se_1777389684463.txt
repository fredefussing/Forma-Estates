import { pipeline } from "@xenova/transformers";

interface VisualTags {
  colors: string[];      // ["beige", "cream", "light-brown"]
  materials: string[];   // ["woven", "rattan", "wood", "textile"]
  style: string[];       // ["scandinavian", "minimalist", "natural"]
  furnitureType: string;  // "lounge_chair", "dining_chair", "sofa"
}

let colorClassifier: any = null;
let styleClassifier: any = null;

export async function analyzeVisualTags(imageBuffer: Buffer): Promise<VisualTags> {
  // Brug CLIP til zero-shot classification af farver
  const colorClassifier = await pipeline(
    "zero-shot-image-classification",
    "Xenova/clip-vit-base-patch32"
  );

  const colors = ["beige", "white", "cream", "light brown", "dark brown", "black", "gray", "blue", "green"];
  const colorResult = await colorClassifier(imageBuffer, colors, { top_k: 3 });
  
  const materials = ["woven rattan", "leather", "fabric", "boucle", "wood", "metal", "velvet"];
  const materialResult = await colorClassifier(imageBuffer, materials, { top_k: 3 });

  const styles = ["scandinavian minimalist", "modern luxury", "rustic", "industrial", "bohemian"];
  const styleResult = await colorClassifier(imageBuffer, styles, { top_k: 2 });

  return {
    colors: colorResult.map((r: any) => r.label),
    materials: materialResult.map((r: any) => r.label),
    style: styleResult.map((r: any) => r.label),
    furnitureType: "unknown" // sættes fra YOLO label
  };
}

import { analyzeVisualTags } from "./analyzeVisual";

export async function findSimilarWithTags(
  cropBuffer: Buffer,
  yoloLabel: string,
  topK: number = 3
) {
  // 1. Analyser visuelle tags fra croppet billede
  const visualTags = await analyzeVisualTags(cropBuffer);
  
  // 2. BYG SQL FILTER — reducer 240k til ~500 relevante
  const colorConditions = visualTags.colors.map(c => 
    `tags @> '[{"color":"${c}"}]'::jsonb OR name ILIKE '%${c}%'`
  ).join(' OR ');
  
  const materialConditions = visualTags.materials.map(m => 
    `tags @> '[{"material":"${m}"}]'::jsonb OR description ILIKE '%${m}%'`
  ).join(' OR ');
  
  const styleConditions = visualTags.style.map(s => 
    `tags @> '[{"style":"${s}"}]'::jsonb OR category ILIKE '%${s}%'`
  ).join(' OR ');

  // 3. PRE-FILTER query — hent kun relevante produkter
  const preFilterSQL = `
    SELECT id, name, image_url, affiliate_link, embedding, tags
    FROM products
    WHERE category ILIKE '%${yoloLabel}%'
      AND (${colorConditions})
      AND (${materialConditions})
      AND (${styleConditions})
    LIMIT 500
  `;
  
  const candidates = await db.query(preFilterSQL);
  
  if (candidates.length === 0) {
    // Fallback: hvis ingen matcher tags, søg bredere
    return findSimilarBasic(cropBuffer, yoloLabel, topK);
  }

  // 4. Generér CLIP vector for crop
  const cropVector = await generateClipVector(cropBuffer);

  // 5. Søg KUN i de 500 filtrerede med pgvector
  const placeholders = candidates.map((_, i) => `$${i + 1}`).join(',');
  const vectorSQL = `
    SELECT id, name, image_url, affiliate_link, price,
           1 - (embedding <=> $${candidates.length + 1}::vector) as similarity
    FROM products
    WHERE id IN (${placeholders})
    ORDER BY embedding <=> $${candidates.length + 1}::vector
    LIMIT $${candidates.length + 2}
  `;
  
  const results = await db.query(vectorSQL, [
    ...candidates.map(c => c.id),
    cropVector,
    topK
  ]);

  return results;
}

interface Detection {
  box: [number, number, number, number]; // x1, y1, x2, y2
  label: string;
  score: number;
}

interface GroupedDetection {
  box: [number, number, number, number];
  label: string;
  score: number;
  count: number; // hvor mange YOLO detektioner er grupperet her
}

export function groupDetections(
  detections: Detection[],
  maxDots: number = 8,
  mergeThreshold: number = 0.3 // 30% overlap = samme møbel
): GroupedDetection[] {
  
  // 1. Filtrér lav confidence
  const filtered = detections.filter(d => d.score >= 0.75);
  
  // 2. Sorter efter score
  filtered.sort((a, b) => b.score - a.score);
  
  const groups: GroupedDetection[] = [];
  
  for (const det of filtered) {
    let merged = false;
    
    // Tjek om denne detection overlapper med eksisterende gruppe
    for (const group of groups) {
      const overlap = calculateIoU(det.box, group.box);
      
      if (overlap > mergeThreshold) {
        // Merge: udvid bounding box til at dække begge
        group.box = mergeBoxes(group.box, det.box);
        group.score = Math.max(group.score, det.score);
        group.count += 1;
        merged = true;
        break;
      }
    }
    
    if (!merged && groups.length < maxDots) {
      groups.push({
        box: det.box,
        label: det.label,
        score: det.score,
        count: 1
      });
    }
  }
  
  // 3. Hvis for få detektioner, tillad lavere confidence
  if (groups.length < 4) {
    const extra = detections
      .filter(d => d.score >= 0.6 && d.score < 0.75)
      .slice(0, maxDots - groups.length);
      
    for (const det of extra) {
      groups.push({ box: det.box, label: det.label, score: det.score, count: 1 });
    }
  }
  
  return groups;
}

function calculateIoU(boxA: number[], boxB: number[]): number {
  const xOverlap = Math.max(0, Math.min(boxA[2], boxB[2]) - Math.max(boxA[0], boxB[0]));
  const yOverlap = Math.max(0, Math.min(boxA[3], boxB[3]) - Math.max(boxA[1], boxB[1]));
  const overlapArea = xOverlap * yOverlap;
  
  const areaA = (boxA[2] - boxA[0]) * (boxA[3] - boxA[1]);
  const areaB = (boxB[2] - boxB[0]) * (boxB[3] - boxB[1]);
  
  return overlapArea / (areaA + areaB - overlapArea);
}

function mergeBoxes(a: number[], b: number[]): [number, number, number, number] {
  return [
    Math.min(a[0], b[0]),
    Math.min(a[1], b[1]),
    Math.max(a[2], b[2]),
    Math.max(a[3], b[3])
  ];
}

export function cropWithPadding(
  imageBuffer: Buffer,
  box: [number, number, number, number], // x1, y1, x2, y2
  paddingPercent: number = 0.15 // 15% padding rundt om møblet
): Buffer {
  const image = sharp(imageBuffer);
  const metadata = await image.metadata();
  
  const width = metadata.width!;
  const height = metadata.height!;
  
  const boxW = box[2] - box[0];
  const boxH = box[3] - box[1];
  
  // Tilføj padding så hele møblet er med + lidt kontekst
  const padX = boxW * paddingPercent;
  const padY = boxH * paddingPercent;
  
  const left = Math.max(0, Math.round(box[0] - padX));
  const top = Math.max(0, Math.round(box[1] - padY));
  const right = Math.min(width, Math.round(box[2] + padX));
  const bottom = Math.min(height, Math.round(box[3] + padY));
  
  return image
    .extract({ left, top, width: right - left, height: bottom - top })
    .resize(224, 224, { fit: 'cover' }) // CLIP input size
    .toBuffer();
}

import { useState, useEffect } from 'react';

interface Props {
  imageUrl: string;
  detections: GroupedDetection[];
  onDotClick: (detection: GroupedDetection) => void;
}

export function FurnitureDetector({ imageUrl, detections, onDotClick }: Props) {
  const [hoveredDot, setHoveredDot] = useState<number | null>(null);
  
  // Max 6-8 prikker — allerede filtreret fra backend
  const visibleDots = detections.slice(0, 8);
  
  return (
    <div className="relative inline-block">
      <img src={imageUrl} alt="AI design" className="w-full rounded-lg" />
      
      {visibleDots.map((det, index) => {
        const centerX = (det.box[0] + det.box[2]) / 2;
        const centerY = (det.box[1] + det.box[3]) / 2;
        
        return (
          <div
            key={index}
            className="absolute transform -translate-x-1/2 -translate-y-1/2 cursor-pointer group"
            style={{ left: `${centerX}%`, top: `${centerY}%` }}
            onClick={() => onDotClick(det)}
            onMouseEnter={() => setHoveredDot(index)}
            onMouseLeave={() => setHoveredDot(null)}
          >
            {/* Pulsating dot */}
            <div className="relative">
              <div className="w-4 h-4 bg-white rounded-full shadow-lg border-2 border-blue-500 animate-pulse" />
              
              {/* Tooltip på hover */}
              {hoveredDot === index && (
                <div className="absolute bottom-full left-1/2 transform -translate-x-1/2 mb-2 
                              bg-black text-white text-xs px-2 py-1 rounded whitespace-nowrap">
                  {det.label} {det.count > 1 ? `(${det.count} fundet)` : ''}

                </div>
              )}
            </div>
          </div>
        );
      })}
      
      {/* Info tekst */}
      <div className="absolute bottom-2 left-2 bg-black/70 text-white text-xs px-2 py-1 rounded">
        {visibleDots.length} møbler fundet — Klik på prikker for at shoppe
      </div>
    </div>
  );
}
