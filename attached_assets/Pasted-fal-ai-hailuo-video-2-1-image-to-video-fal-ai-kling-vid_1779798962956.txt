// ============================================
// 1. 3D PLANTEGNING (Billede)
// ============================================
export async function generate3DFloorplan(
  prompt: string,
  aspectRatio: string = "16:9"
): Promise<{ imageUrl: string }> {
  const result = await fal.subscribe("fal-ai/nano-banana-2", {
    input: {
      prompt: `Isometric 3D architectural floor plan visualization, ${prompt}, viewed from a 35-degree elevated angle (axonometric perspective, NOT directly overhead), natural daylight, soft shadows, architectural cutaway style showing interior details, photorealistic 3D render, 8k, ultra-detailed`,
      aspect_ratio: aspectRatio as any,
      resolution: "2K",
    },
  });

  const imageUrl = (result.data as any).images?.[0]?.url;
  if (!imageUrl) throw new Error("No image generated");
  return { imageUrl };
}

// ============================================
// 2. FØR/EFTER ANIMATIONSVIDEO — KLING 3.0 PRO
// ============================================
export async function generateAnimationVideo(
  beforeImageUrl: string,
  _afterImageUrl: string,
  roomType: string = "room"
): Promise<{ videoUrl: string }> {
  // KLING 3.0 PRO — image-to-video med start + end frame
  const result = await fal.subscribe(
    "fal-ai/kling-video/v3/pro/image-to-video",
    {
      input: {
        image_url: beforeImageUrl,
        prompt: `Cinematic renovation transformation of a ${roomType}. The camera slowly pushes forward into the space as the entire room magically transforms — walls change color, flooring spreads seamlessly, new furniture materializes in place, lighting fixtures illuminate the space with warm golden light. The transformation is smooth, elegant, and weightless — no abrupt cuts. Natural daylight floods through windows gradually increasing in brightness. Movement is slow and cinematic, like a professional real estate walkthrough video. Photorealistic, 8k quality, architectural visualization.`,
        end_image_url: _afterImageUrl,
        duration: 8,
        aspect_ratio: "16:9",
      },
    }
  );

  const videoUrl = (result.data as any).video?.url;
  if (!videoUrl) throw new Error("No video generated");
  return { videoUrl };
}
