        <div class="actions">
            <button class="btn-primary" onclick="generateDesign()">Generér mit design</button>
        </div>
    </div>

    <script>
        const urlParams = new URLSearchParams(window.location.search);
        const room = urlParams.get('room') || 'Stue';
        const style = urlParams.get('style') || 'Skandinavisk';
        const budget = urlParams.get('budget') || '25000';

        document.getElementById('selectedRoom').textContent = room;
        document.getElementById('selectedStyle').textContent = style;
        document.getElementById('selectedBudget').textContent = 
            budget === '10000' ? '10.000 kr' : 
            budget === '25000' ? '25.000 kr' : '50.000+ kr';

        function generateDesign() {
            let promptText = `${style} ${room}, `;
            
            if (budget === '10000') promptText += 'budget-friendly, IKEA level, simple furniture';
            else if (budget === '25000') promptText += 'mid-range quality, stylish furniture';
            else promptText += 'luxury designer furniture, high-end materials';

            window.location.href = `/generer?room=${encodeURIComponent(room)}&style=${encodeURIComponent(style)}&budget=${budget}&prompt=${encodeURIComponent(promptText)}`;
        }
    </script>
