Fejlmeddelelse: Failed to execute 'json' on 'Response': Unexpected token '<', "<!DOCTYPE "... is not valid JSON

Dette betyder: Din POST /api/bolig/cases returnerer HTML, ikke JSON.

Tilføj DETTE i din frontend mutation (før res.json()):

mutationFn: async ({ address, caseNo, notes }) => {
  const token = await auth.currentUser?.getIdToken();
  console.log("🔑 Token:", token ? "Tilstede" : "MANGLER");

      caseNo, 
      notes, 
      marketDateISO: new Date().toISOString().split('T')[0] 
    }),
  });
  
  console.log("📡 Response status:", res.status);
  console.log("📡 Response headers:", Object.fromEntries(res.headers));
  
  // Tjek om det er JSON eller HTML
  const contentType = res.headers.get("content-type");
  console.log("📡 Content-Type:", contentType);
  
  if (!contentType?.includes("application/json")) {
    const text = await res.text();
    console.error("📡 Response er IKKE JSON:");
    console.error(text.substring(0, 500));  // Vis første 500 chars
    throw new Error(`Server returnerede ${res.status} med ${contentType}, ikke JSON`);
  }
  
  return res.json();
},

ERSTAT med eksplicit JSON middleware sikring:

app.post("/api/bolig/cases", async (req, res) => {
  try {
    // Sikr at vi altid returnerer JSON
    res.setHeader('Content-Type', 'application/json');
    
    const authHeader = req.headers.authorization;
    if (!authHeader) {
      return res.status(401).json({ error: "Missing authorization header" });
    }
    
    const user = await verifyFirebaseToken(authHeader);
    const dbUser = await storage.getUserByFirebaseUid(user.uid);
    
    if (!dbUser) {
      return res.status(401).json({ error: "User not found in database" });
    }
    
    const { address, caseNo, notes, marketDateISO } = req.body;
    
    if (!address || !address.trim()) {
      return res.status(400).json({ error: "Address is required" });
    }
    
    const newCase = await storage.createBoligCase({
      userId: dbUser.id,
      address: address.trim(),
      caseNo: caseNo?.trim() || null,
      notes: notes?.trim() || null,
      status: "active",
      marketDateISO: marketDateISO || new Date().toISOString().split('T')[0],
    });
    
    return res.status(201).json(newCase);
    
  } catch (error) {
    console.error("❌ POST /api/bolig/cases fejl:", error);
    // VIKTIG: Returner alltid JSON, aldri HTML
    return res.status(500).json({ 
      error: "Internal server error",
      message: error instanceof Error ? error.message : "Unknown error"
    });
  }
});
