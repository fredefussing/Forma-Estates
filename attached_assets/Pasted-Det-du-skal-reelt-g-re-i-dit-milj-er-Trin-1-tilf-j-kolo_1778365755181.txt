🧩 Trin 2 – lav batchscriptet til text‑embeddings
Opret filen scripts/generate‑text‑vectors.mjs

js


import { pipeline } from "@xenova/transformers";
import pg from "pg";
const client = new pg.Client({ connectionString: process.env.DATABASE_URL });
await client.connect();
const encoder = await pipeline("feature-extraction", "Xenova/clip-vit-base-patch32");
const BATCH = 500;
while (true) {
  const { rows } = await client.query(
    "SELECT id, name, tags FROM products WHERE vector_text IS NULL LIMIT $1",
    [BATCH]
  );
  if (rows.length === 0) break;
  for (const row of rows) {
    const tags =
      typeof row.tags === "object"
        ? row.tags
        : row.tags
        ? JSON.parse(row.tags)
        : {};
    const prompt = `${row.name}, ${tags.style || ""}, ${tags.color || ""}, ${
      tags.material || ""
    }`.trim();
    const out = await encoder(prompt);
    const vec = Array.from(out.data[0]);
    const norm = Math.hypot(...vec);
    const normalized = vec.map((v) => v / norm);
    await client.query("UPDATE products SET vector_text=$1 WHERE id=$2", [
      normalized,
      row.id,
    ]);
  }
  console.log(`Processed ${rows.length}...`);
}
console.log("✅ text embeddings done!");
await client.end();
