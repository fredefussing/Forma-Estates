import { batchTagProducts } from "../server/tagProducts";
import { db } from "../server/db";

/**
 * Batch tag top produkter
 * Kør med: npx ts-node scripts/tag-top-products.ts
 * Eller: node --loader ts-node/esm scripts/tag-top-products.ts
 */

async function main() {
  const TARGET = 10000; // Tag top 10.000
  const BATCH_SIZE = 100; // 100 ad gangen

  console.log("=== TAGGING SYSTEM ===");
  console.log(`Target: ${TARGET} products`);
  console.log(`Batch size: ${BATCH_SIZE}`);
  console.log("");

  // Tjek hvor mange der allerede er tagged
  const alreadyTagged = await db.query(`
    SELECT COUNT(*) as count FROM products WHERE tag_processed = TRUE
  `);

  const totalUntagged = await db.query(`
    SELECT COUNT(*) as count FROM products WHERE tag_processed = FALSE
  `);

  console.log(`Already tagged: ${alreadyTagged[0].count}`);
  console.log(`Untagged: ${totalUntagged[0].count}`);
  console.log("");

  let totalSuccess = 0;
  let totalFailed = 0;
  let offset = 0;

  while (totalSuccess + totalFailed < TARGET) {
    const remaining = TARGET - (totalSuccess + totalFailed);
    const batchSize = Math.min(BATCH_SIZE, remaining);

    console.log(`\n--- Batch ${Math.floor(offset / BATCH_SIZE) + 1} ---`);
    console.log(`Offset: ${offset}, Size: ${batchSize}`);

    const result = await batchTagProducts(batchSize, offset);

    totalSuccess += result.success;
    totalFailed += result.failed;
    offset += batchSize;

    console.log(`Success: ${result.success}, Failed: ${result.failed}`);
    console.log(`Total progress: ${totalSuccess + totalFailed}/${TARGET}`);
    console.log(`Success rate: ${((totalSuccess / (totalSuccess + totalFailed)) * 100).toFixed(1)}%`);

    // Progress every 500
    if ((totalSuccess + totalFailed) % 500 === 0) {
      const stats = await db.query(`
        SELECT 
          tags->>'style' as style,
          COUNT(*) as count
        FROM products 
        WHERE tag_processed = TRUE
        GROUP BY tags->>'style'
        ORDER BY count DESC
        LIMIT 5
      `);

      console.log("\nTop styles so far:");
      stats.forEach((s: any) => console.log(`  ${s.style}: ${s.count}`));
    }

    // Auto-pause every 1000 to avoid rate limits
    if ((totalSuccess + totalFailed) % 1000 === 0 && remaining > 0) {
      console.log("\n--- Pausing 60s to avoid rate limits ---");
      await new Promise(r => setTimeout(r, 60000));
    }
  }

  console.log("\n=== DONE ===");
  console.log(`Total tagged: ${totalSuccess + totalFailed}`);
  console.log(`Success: ${totalSuccess}`);
  console.log(`Failed: ${totalFailed}`);
  console.log(`Rate: ${((totalSuccess / (totalSuccess + totalFailed)) * 100).toFixed(1)}%`);

  // Final stats
  const finalStats = await db.query(`
    SELECT 
      tags->>'style' as style,
      COUNT(*) as count,
      AVG(tag_confidence)::float as avg_confidence
    FROM products 
    WHERE tag_processed = TRUE
    GROUP BY tags->>'style'
    ORDER BY count DESC
  `);

  console.log("\nFinal style distribution:");
  finalStats.forEach((s: any) => {
    console.log(`  ${s.style}: ${s.count} (avg confidence: ${(s.avg_confidence * 100).toFixed(0)}%)`);
  });

  process.exit(0);
}

main().catch(e => {
  console.error("Fatal error:", e);
  process.exit(1);
});
