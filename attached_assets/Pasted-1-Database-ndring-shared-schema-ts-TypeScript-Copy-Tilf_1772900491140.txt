app.get("/api/designs/my", async (req, res) => {
  try {
    const decoded = await verifyFirebaseToken(req);
    const [user] = await db.select().from(users).where(eq(users.firebaseUid, decoded.uid));
    
    if (!user) {
      return res.status(404).json({ error: "Bruger ikke fundet" });
    }
    
    // Kun designs hvor userId matcher
    const myDesigns = await db
      .select()
      .from(designs)
      .where(eq(designs.userId, user.id))
      .orderBy(desc(designs.createdAt)); // nyeste først
    
    res.json(myDesigns);
  } catch (error) {
    res.status(401).json({ error: "Ugyldig token" });
  }
});

// Gem design med userId
app.post("/api/designs", async (req, res) => {
  try {
    const decoded = await verifyFirebaseToken(req);
    const { imageUrl, originalImageUrl, roomType, style, budget, tier } = req.body;
    
    const [user] = await db.select().from(users).where(eq(users.firebaseUid, decoded.uid));
    
    if (!user || user.creditsRemaining <= 0) {
      return res.status(403).json({ error: "Ingen credits" });
    }
    
    // Træk credit
    await db.update(users)
      .set({ creditsRemaining: user.creditsRemaining - 1 })
      .where(eq(users.id, user.id));
    
    // Gem design med userId
    const [design] = await db.insert(designs).values({
      userId: user.id, // ← VIGTIGT: Kobler til bruger
      imageUrl,
      originalImageUrl,
      roomType,
      style,
      budget,
      tier,
      status: "pending",
    }).returning();
    
    res.json({ success: true, designId: design.id });
  } catch (error) {
    res.status(500).json({ error: "Fejl" });
  }
});

import { useEffect, useState } from "react";
import { useAuth } from "@/hooks/useAuth";
import { Link } from "wouter";

export default function MyDesigns() {
  const { user } = useAuth();
  const [designs, setDesigns] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fetchMyDesigns = async () => {
      const token = await user?.getIdToken();
      const res = await fetch("/api/designs/my", {
        headers: { Authorization: `Bearer ${token}` }
      });
      const data = await res.json();
      setDesigns(data);
      setLoading(false);
    };
    
    if (user) fetchMyDesigns();
  }, [user]);

  if (loading) return <div>Loader...</div>;

  return (
    <div className="p-6 max-w-6xl mx-auto">
      <h1 className="text-3xl font-bold mb-6">Dine tidligere designs</h1>
      
      {designs.length === 0 ? (
        <p>Du har ikke lavet nogen designs endnu.</p>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
          {designs.map((design) => (
            <Link key={design.id} href={`/design/${design.id}`}>
              <div className="border rounded-lg overflow-hidden hover:shadow-lg cursor-pointer">
                <img 
                  src={design.imageUrl} 
                  alt={design.style}
                  className="w-full h-48 object-cover"
                />
                <div className="p-4">
                  <p className="font-semibold">{design.roomType} - {design.style}</p>
                  <p className="text-sm text-gray-500">
                    {new Date(design.createdAt).toLocaleDateString('da-DK')}
                  </p>
                  <p className="text-sm text-gray-500">
                    Budget: {design.budget} kr
                  </p>
                </div>
              </div>
            </Link>
          ))}
        </div>
      )}
    </div>
  );
}

import { useEffect, useState } from "react";
import { useRoute, useLocation } from "wouter";
import { useAuth } from "@/hooks/useAuth";
import BeforeAfterSlider from "@/components/before-after-slider";

export default function DesignDetail() {
  const [match, params] = useRoute("/design/:id");
  const { user } = useAuth();
  const [design, setDesign] = useState(null);
  const [loading, setLoading] = useState(true);
  const [, setLocation] = useLocation();

  useEffect(() => {
    const fetchDesign = async () => {
      const token = await user?.getIdToken();
      const res = await fetch(`/api/designs/${params.id}`, {
        headers: { Authorization: `Bearer ${token}` }
      });
      
      if (res.status === 403) {
        // Ikke deres design
        setLocation("/my-designs");
        return;
      }
      
      const data = await res.json();
      setDesign(data);
      setLoading(false);
    };
    
    if (user) fetchDesign();
  }, [user, params.id]);

  if (loading) return <div>Loader...</div>;
  if (!design) return <div>Design ikke fundet</div>;

  return (
    <div className="p-6 max-w-4xl mx-auto">
      <h1 className="text-2xl font-bold mb-4">
        {design.roomType} - {design.style}
      </h1>
      
      {/* Før/Efter Slider */}
      <div className="mb-6">
        <BeforeAfterSlider 
          before={design.originalImageUrl || design.imageUrl}
          after={design.imageUrl}
        />
      </div>
      
      <div className="grid grid-cols-2 gap-4 text-sm">
        <div>
          <p className="font-semibold">Rum:</p>
          <p>{design.roomType}</p>
        </div>
        <div>
          <p className="font-semibold">Stil:</p>
          <p>{design.style}</p>
        </div>
        <div>
          <p className="font-semibold">Budget:</p>
          <p>{design.budget} kr</p>
        </div>
        <div>
          <p className="font-semibold">Dato:</p>
          <p>{new Date(design.createdAt).toLocaleDateString('da-DK')}</p>
        </div>
      </div>
    </div>
  );
}

// Hent specifikt design - kun hvis det tilhører brugeren
app.get("/api/designs/:id", async (req, res) => {
  try {
    const decoded = await verifyFirebaseToken(req);
    const designId = parseInt(req.params.id);
    
    const [user] = await db.select().from(users).where(eq(users.firebaseUid, decoded.uid));
    
    if (!user) {
      return res.status(404).json({ error: "Bruger ikke fundet" });
    }
    
    const [design] = await db
      .select()
      .from(designs)
      .where(eq(designs.id, designId))
      .limit(1);
    
    if (!design) {
      return res.status(404).json({ error: "Design ikke fundet" });
    }
    
    // Tjek om designet tilhører brugeren
    if (design.userId !== user.id) {
      return res.status(403).json({ error: "Adgang nægtet" });
    }
    
    res.json(design);
  } catch (error) {
    res.status(500).json({ error: "Server fejl" });
  }
});
