// ===== 1. 3D PLANTEGNING =====
export async function generate3DFloorplan(
  prompt: string,
  aspectRatio: string = "16:9"
): Promise<{ imageUrl: string }> {
  const result = await fal.subscribe("fal-ai/nano-banana-2", {
    input: {
      prompt: `Isometric 3D architectural floor plan visualization, ${prompt}, viewed from a 35-degree elevated angle (axonometric perspective, NOT directly overhead), natural daylight, soft shadows, architectural cutaway style showing interior details, photorealistic 3D render, 8k, ultra-detailed`,
      aspect_ratio: aspectRatio as any,
      resolution: "2K",
    },
  });

  const imageUrl = (result.data as any).images?.[0]?.url;
  if (!imageUrl) throw new Error("No image generated");
  return { imageUrl };
}

// ===== 2. FØR/EFTER ANIMATIONSVIDEO =====
export async function generateAnimationVideo(
  beforeImageUrl: string,
  roomType: string = "room"
): Promise<{ videoUrl: string }> {
  const result = await fal.subscribe("fal-ai/hailuo/video-2.1/image-to-video", {
    input: {
      image_url: beforeImageUrl,
      prompt: `Magical renovation transformation of a ${roomType}. The space transforms from its original state into a beautifully renovated version. Elements materialize and grow into place smoothly — cabinetry rises from the floor, countertops expand outward, lighting fixtures descend from the ceiling, walls change color, flooring spreads seamlessly. Warm natural daylight floods the space. Smooth, slow, cinematic motion, no abrupt movement, photorealistic, 8k quality.`,
      duration: 6,
    },
  });

  const videoUrl = (result.data as any).video?.url;
  if (!videoUrl) throw new Error("No video generated");
  return { videoUrl };
}

import { z } from "zod";
import { createRouter, publicQuery } from "../middleware";
import { getDb } from "../queries/connection";
import { renovations } from "@db/schema";
import { eq } from "drizzle-orm";
import { generateAnimationVideo } from "../lib/fal";

export const renovationRouter = createRouter({
  create: publicQuery
    .input(
      z.object({
        name: z.string().min(1),
        beforeImageUrl: z.string().url(),
        afterImageUrl: z.string().url(),
        roomType: z.string().default("room"),
      })
    )
    .mutation(async ({ input }) => {
      const db = getDb();
      const [renovation] = await db.insert(renovations).values({
        name: input.name,
        beforeImageUrl: input.beforeImageUrl,
        afterImageUrl: input.afterImageUrl,
        roomType: input.roomType,
        status: "pending",
      }).$returningId();

      return { id: renovation.id };
    }),

  generateVideo: publicQuery
    .input(z.object({ id: z.number() }))
    .mutation(async ({ input }) => {
      const db = getDb();
      const [renovation] = await db
        .select()
        .from(renovations)
        .where(eq(renovations.id, input.id));

      if (!renovation) return { success: false, error: "Not found" };

      await db
        .update(renovations)
        .set({ status: "generating" })
        .where(eq(renovations.id, input.id));

      try {
        const result = await generateAnimationVideo(
          renovation.beforeImageUrl,
          renovation.roomType || "room"
        );

        await db
          .update(renovations)
          .set({ videoUrl: result.videoUrl, status: "completed" })
          .where(eq(renovations.id, input.id));

        return { success: true, videoUrl: result.videoUrl };
      } catch (error: any) {
        await db
          .update(renovations)
          .set({ status: "failed" })
          .where(eq(renovations.id, input.id));
        return { success: false, error: error.message };
      }
    }),

  list: publicQuery.query(async () => {
    const db = getDb();
    return db.select().from(renovations).orderBy(renovations.createdAt);
  }),

  get: publicQuery
    .input(z.object({ id: z.number() }))
    .query(async ({ input }) => {
      const db = getDb();
      const [renovation] = await db
        .select()
        .from(renovations)
        .where(eq(renovations.id, input.id));
      return renovation || null;
    }),

  delete: publicQuery
    .input(z.object({ id: z.number() }))
    .mutation(async ({ input }) => {
      const db = getDb();
      await db.delete(renovations).where(eq(renovations.id, input.id));
      return { success: true };
    }),
});

import { useState } from "react";
import { trpc } from "@/providers/trpc";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Loader2, Film, Sparkles, ArrowLeft } from "lucide-react";
import { useNavigate } from "react-router";

const ROOM_TYPES = [
  { id: "kitchen", label: "Køkken" },
  { id: "living room", label: "Stue" },
  { id: "bathroom", label: "Badeværelse" },
  { id: "bedroom", label: "Soveværelse" },
  { id: "apartment", label: "Hele lejligheden" },
];

export default function AnimatedVideos() {
  const navigate = useNavigate();
  const [name, setName] = useState("");
  const [beforeUrl, setBeforeUrl] = useState("");
  const [afterUrl, setAfterUrl] = useState("");
  const [roomType, setRoomType] = useState("kitchen");
  const [videoUrl, setVideoUrl] = useState<string | null>(null);
  const [isGenerating, setIsGenerating] = useState(false);
  const [status, setStatus] = useState("");

  const createRenovation = trpc.renovation.create.useMutation();
  const generateVideo = trpc.renovation.generateVideo.useMutation();

  const handleGenerate = async () => {
    if (!name.trim() || !beforeUrl.trim() || !afterUrl.trim()) return;
    setIsGenerating(true);
    setStatus("Gemmer projekt...");

    try {
      const project = await createRenovation.mutateAsync({
        name: name.trim(),
        beforeImageUrl: beforeUrl.trim(),
        afterImageUrl: afterUrl.trim(),
        roomType,
      });

      setStatus("Genererer animationsvideo med fal.ai... (1-3 minutter)");
      const result = await generateVideo.mutateAsync({ id: project.id });

      if (result.success && result.videoUrl) {
        setVideoUrl(result.videoUrl);
        setStatus("Færdig!");
      } else {
        setStatus(result.error || "Fejl");
      }
    } catch (error: any) {
      setStatus(error.message || "Fejl");
    }
    setIsGenerating(false);
  };

  return (
    <div className="min-h-screen bg-background">
      {/* Header */}
      <header className="fixed top-0 left-0 right-0 z-50 bg-background/90 backdrop-blur-xl border-b border-border">
        <div className="max-w-4xl mx-auto h-14 flex items-center gap-4 px-4">
          <button onClick={() => navigate("/")} className="p-2 rounded-lg hover:bg-muted text-muted-foreground">
            <ArrowLeft size={18} />
          </button>
          <Film size={20} className="text-primary" />
          <h1 className="text-sm font-medium text-foreground">Animerede Videoer</h1>
        </div>
      </header>

      <div className="pt-24 pb-16 px-4">
        <div className="max-w-2xl mx-auto">
          {/* Intro */}
          <div className="text-center mb-8">
            <h2 className="font-display text-3xl text-foreground mb-2">Animerede Videoer</h2>
            <p className="text-muted-foreground text-sm">
              Upload før- og efter-billeder — AI skaber en magisk transformationsvideo
            </p>
          </div>

          <div className="p-6 rounded-2xl border border-border bg-card space-y-5">
            {/* Name */}
            <div>
              <Label className="text-sm font-medium text-foreground mb-2 block">Projektnavn</Label>
              <Input
                value={name}
                onChange={(e) => setName(e.target.value)}
                placeholder="f.eks. Køkkenrenovering Sønderskov"
                className="bg-secondary border-border text-foreground"
              />
            </div>

            {/* Image URLs */}
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              <div>
                <Label className="text-sm font-medium text-foreground mb-2 block">Før-billede URL</Label>
                <Input
                  value={beforeUrl}
                  onChange={(e) => setBeforeUrl(e.target.value)}
                  placeholder="https://... før renovering"
                  className="bg-secondary border-border text-foreground"
                />
              </div>
