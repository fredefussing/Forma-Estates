// /src/components/RotatingStats.tsx

import { useState, useEffect } from 'react';

export function RotatingStats() {
  const stats = [
    { icon: '🔥', value: '12', label: 'designs i dag' },
    { icon: '⭐', value: '4.8', label: 'stjerner' },
    { icon: '💰', value: '8.7k', label: 'kr sparet' }
  ];
  
  const [currentIndex, setCurrentIndex] = useState(0);
  
  useEffect(() => {
    const interval = setInterval(() => {
      setCurrentIndex((prev) => (prev + 1) % stats.length);
    }, 4000);
    
    return () => clearInterval(interval);
  }, []);
  
  const current = stats[currentIndex];
  
  return (
    <div className="rotating-stats">
      <span className="stat-icon">{current.icon}</span>
      <span className="stat-value">{current.value}</span>
      <span className="stat-label">{current.label}</span>
    </div>
  );
}
