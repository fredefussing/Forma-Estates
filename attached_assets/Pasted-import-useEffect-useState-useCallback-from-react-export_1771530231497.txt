import { useEffect, useState, useCallback } from 'react';

export function useTransformationJob(jobId: string | null) {
  const [status, setStatus] = useState<'idle' | 'pending' | 'processing' | 'completed' | 'failed'>('idle');
  const [resultUrl, setResultUrl] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [progress, setProgress] = useState(0);

  const reset = useCallback(() => {
    setStatus('idle');
    setResultUrl(null);
    setError(null);
    setProgress(0);
  }, []);

  useEffect(() => {
    if (!jobId) {
      reset();
      return;
    }

    setStatus('pending');
    let isActive = true; // For at undgå memory leaks ved unmount
    let attempts = 0;
    const maxAttempts = 60; // 3 minutter max (60 * 3 sekunder)
    
    const poll = async () => {
      while (isActive && attempts < maxAttempts) {
        attempts++;
        setProgress(Math.min((attempts / 20) * 100, 95)); // Fake progress til 95%
        
        try {
          const response = await fetch(`/api/transform/${jobId}/status`);
          const data = await response.json();
          
          if (!isActive) break; // Stop hvis komponent er unmounted

          if (data.status === 'completed') {
            setStatus('completed');
            setResultUrl(data.resultUrl);
            setProgress(100);
            return; // Stop polling
          }
          
          if (data.status === 'failed' || data.status === 'error') {
            setStatus('failed');
            setError(data.error || 'Noget gik galt. Prøv igen.');
            return; // Stop polling
          }
          
          // Fortsæt polling for pending/processing
          await new Promise(resolve => setTimeout(resolve, 3000));
          
        } catch (err) {
          console.error('Polling fejl:', err);
          // Fortsæt polling ved netværksfejl, men tæl op
          await new Promise(resolve => setTimeout(resolve, 3000));
        }
      }
      
      // Timeout efter maxAttempts
      if (isActive && attempts >= maxAttempts) {
        setStatus('failed');
        setError('Det tog for lang tid. Prøv igen eller kontakt support.');
      }
    };

    poll();
    
    // Cleanup function - VIKTIG ved refresh
    return () => {
      isActive = false;
    };
  }, [jobId, reset]);

  return { status, resultUrl, error, progress, reset };
}
