function buildPrompt(roomType, tier, includePlants) {
  // Lookup det rigtige vocab fra tabellen ovenfor
  const vocab = SCANDINAVIAN_VOCAB[roomType][tier];
  
  let prompt = `Completely redesign this ${roomType}. ${vocab}`;
  
  if (includePlants) {
    prompt += " Green indoor plants.";
  }
  
  prompt += " Replace all existing furniture and decor.";
  
  return prompt;
}

async function generateDesign(uploadedFile, roomType, tier, includePlants) {
  // 1. Gem fil lokalt med Multer
  //    Path: /uploads/filename.jpg
  //    Public URL: https://{REPLIT_DOMAIN}/uploads/filename.jpg
  
  const publicUrl = `https://${process.env.REPLIT_DOMAIN}/uploads/${filename}`;
  
  // 2. Byg prompt
  const prompt = buildPrompt(roomType, tier, includePlants);
  
  // 3. Send til Collov
  const form = new FormData();
  form.append('uploadUrl', publicUrl);
  form.append('prompt', prompt);
  
  const sendRes = await fetch(
    'https://api.collov.ai/flair/enterpriseApi/edit/generate',
    {
      method: 'POST',
      headers: { 'apiKey': process.env.COLOV_API_KEY },
      body: form,
    }
  );
  
  const { data: { uuid } } = await sendRes.json();
  
  // 4. Poll
  for (let i = 0; i < 60; i++) {
    await new Promise(r => setTimeout(r, 3000));
    const pollRes = await fetch(
      `https://api.collov.ai/flair/enterpriseApi/edit/getRecord?uuid=${uuid}`,
      { headers: { 'apiKey': process.env.COLOV_API_KEY } }
    );
    const pollData = await pollRes.json();
    if (pollData.data.status === 'SUCCESS') return pollData.data.generateUrl;
    if (pollData.data.status === 'FAILED') throw new Error('Failed');
  }
  
  throw new Error('Timeout');
}
