function getPrice(product, qty) {
  if (qty === 0) return 0;
  const tier = TIERS[product].find(t => qty <= t.max);
  return tier.price;
}

app.post('/api/calculate-package', (req, res) => {
  const { aiVisual = 0, plan3d = 0, transformVideo = 0, showcase = 0 } = req.body;
  
  const items = [];
  let originalTotal = 0;
  let grandTotal = 0;
  
  const addItem = (key, name, basePrice) => {
    const qty = req.body[key];
    if (qty > 0) {
      const unitPrice = getPrice(key, qty);
      const total = unitPrice * qty;
      const original = basePrice * qty;
      items.push({ name, quantity: qty, unitPrice, originalUnitPrice: basePrice, total, originalTotal: original, discount: original - total });
      originalTotal += original;
      grandTotal += total;
    }
  };
  
  addItem('aiVisual', 'AI Visualisering', 100);
  addItem('plan3d', '3D Plantegning', 300);
  addItem('transformVideo', 'Transformering Video', 300);
  addItem('showcase', 'Bolig Showcase Video', 500);
  
  res.json({
    items,
    originalTotal,
    grandTotal,
    totalSavings: originalTotal - grandTotal,
    totalDiscountPercent: originalTotal > 0 ? Math.round(((originalTotal - grandTotal) / originalTotal) * 100) : 0,
  });
});

app.post('/api/create-package-checkout', async (req, res) => {
  const { aiVisual = 0, plan3d = 0, transformVideo = 0, showcase = 0, customerEmail } = req.body;
  
  const calcRes = await fetch(`http://localhost:${process.env.PORT || 5000}/api/calculate-package`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ aiVisual, plan3d, transformVideo, showcase }),
  });
  const calc = await calcRes.json();
  
  const line_items = [];
  
  for (const item of calc.items) {
    line_items.push({
      price_data: {
        currency: 'dkk',
        product_data: { name: `${item.name} (${item.quantity} stk)` },
        unit_amount: item.unitPrice * 100,
      },
      quantity: item.quantity,
    });
  }
  
  if (calc.totalSavings > 0) {
    line_items.push({
      price_data: {
        currency: 'dkk',
        product_data: { name: 'Samlet rabat' },
        unit_amount: -calc.totalSavings * 100,
      },
      quantity: 1,
    });
  }
  
  const session = await stripe.checkout.sessions.create({
    mode: 'payment',
    customer_email: customerEmail,
    line_items,
    success_url: 'https://formaestates.com/success?session_id={CHECKOUT_SESSION_ID}',
    cancel_url: 'https://formaestates.com/pakke',
    metadata: {
      type: 'custom_package',
      ai_visual: aiVisual,
      plan_3d: plan3d,
      transform_video: transformVideo,
      showcase: showcase,
    },
  });
  
  res.json({ url: session.url });
});

app.post('/api/create-subscription-checkout', async (req, res) => {
  const { priceId, customerEmail } = req.body;
  
  const session = await stripe.checkout.sessions.create({
    mode: 'subscription',
    customer_email: customerEmail,
    line_items: [{ price: priceId, quantity: 1 }],
    success_url: 'https://formaestates.com/success?session_id={CHECKOUT_SESSION_ID}',
    cancel_url: 'https://formaestates.com/pricing',
    metadata: { type: 'subscription' },
  });
  
  res.json({ url: session.url });
});

// ============================================================
// 4. HENT KVITTERING
// ============================================================
app.get('/api/session-details/:sessionId', async (req, res) => {
  const session = await stripe.checkout.sessions.retrieve(
    req.params.sessionId,
    { expand: ['line_items'] }
  );
  
  res.json({
    status: session.payment_status,
    amount_total: session.amount_total,
    amount_subtotal: session.amount_subtotal,
    tax: session.total_details?.amount_tax,
    customer_email: session.customer_email,
    metadata: session.metadata,
  });
});
