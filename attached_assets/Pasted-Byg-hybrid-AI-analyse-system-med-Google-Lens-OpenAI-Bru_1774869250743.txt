import axios from 'axios';

export async function searchWithGoogleLens(imageUrl: string) {
  try {
    const response = await axios.get('https://serpapi.com/search', {
      params: {
        engine: 'google_lens',
        url: imageUrl,
        api_key: process.env.SERPAPI_KEY,
        country: 'dk',
        hl: 'da'
      },
      timeout: 10000
    });
    
    // Filtrer shopping results med priser fra kendte butikker
    const shoppingResults = response.data.visual_matches?.filter((item: any) => {
      const hasPrice = item.price && item.price.value;
      const knownStore = ['IKEA', 'JYSK', 'ILVA', 'Bolia', 'HAY', 'Bahne', 'Menu', 'Nordic Dream', 'Drømmeland'].some(
        store => item.source?.toLowerCase().includes(store.toLowerCase())
      );
      return hasPrice && (knownStore || item.source);
    }) || [];
    
    // Sortér efter pris-match (lavest først) og tag top 3
    return shoppingResults
      .sort((a: any, b: any) => parseFloat(a.price.value) - parseFloat(b.price.value))
      .slice(0, 3);
      
  } catch (error) {
    console.error('Google Lens error:', error);
    return [];
  }
}

TRIN 2: Opdater server/ai_analyzer.ts

export async function analyzeDesignImage(imageUrl: string, budget: number, roomType: string, style: string) {
  // Prøv Google Lens først
  const googleLensResults = await searchWithGoogleLens(imageUrl);
  
  let products;
  let method;
  
  if (googleLensResults.length > 0) {
    // Google Lens fandt produkter
    products = googleLensResults.map((item: any) => ({
      name: item.title,
      source: item.source,
      price: parseFloat(item.price.value),
      currency: item.price.currency,
      link: item.link,
      thumbnail: item.thumbnail,
      inStock: item.in_stock,
      method: 'google_lens'
    }));
    method = 'google_lens';
  } else {
    // Fallback til OpenAI
    const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
    const response = await openai.chat.completions.create({
      model: "gpt-4o",
      messages: [{
        role: "user",
        content: [{
          type: "text",
          text: `Analyser dette ${roomType} i ${style} stil. Budget: ${budget} DKK. 
          Fordel 85% på produkter. Identificer møbler. 
          Returner JSON med products (name, searchTerms, exactBudget, visualDescription, recommendedStores[2]).`
        }, {
          type: "image_url",
          image_url: { url: imageUrl, detail: "high" }
        }]
      }],
      response_format: { type: "json_object" }
    });
    
    const aiResult = JSON.parse(response.choices[0].message.content);
    products = aiResult.products.map((p: any) => ({ ...p, method: 'openai' }));
    method = 'openai_fallback';
  }
  
  // Beregn profit (15%)
  const totalProductBudget = products.reduce((sum: number, p: any) => sum + (p.price || p.exactBudget), 0);
  const profit = Math.round(totalProductBudget * 0.176); // 15% af total = 85% produkter
  
  return { products, totalProductBudget, profit, method };
}
