// /src/components/CustomNotes.tsx

import { useState } from 'react';

interface CustomNotesProps {
  value: string;
  onChange: (value: string) => void;
  maxLength?: number;
}

export function CustomNotes({ value, onChange, maxLength = 100 }: CustomNotesProps) {
  const [isExpanded, setIsExpanded] = useState(false);
  
  const examples = [
    "Mørkegrøn accent-væg bag sengen",
    "Behold eksisterende sofa, skift resten",
    "Tilføj planter og varme trætoner",
    "Sort sengelampe og minimalistisk look"
  ];
  
  return (
    <div className="custom-notes">
      <button 
        type="button"
        onClick={() => setIsExpanded(!isExpanded)}
        className="toggle-button"
      >
        {isExpanded ? '−' : '+'} Har du specifikke ønsker? (valgfrit)
      </button>
      
      {isExpanded && (
        <div className="notes-content">
          <p className="helper-text">
            Beskriv detaljer vi skal tage hensyn til:
          </p>
          
          <div className="examples">
            {examples.map((ex, i) => (
              <button
                key={i}
                type="button"
                className="example-chip"
                onClick={() => onChange(ex)}
              >
                "{ex}"
              </button>
            ))}
          </div>
          
          <div className="input-wrapper">
            <textarea
              value={value}
              onChange={(e) => onChange(e.target.value.slice(0, maxLength))}
              placeholder="Dine ønsker..."
              rows={3}
              className="notes-textarea"
            />
            <span className={`char-count ${value.length >= maxLength ? 'limit' : ''}`}>
              {value.length}/{maxLength}
            </span>
          </div>
          
          {value.length > 0 && (
            <p className="disclaimer">
              Vi bestræber os på at opfylde dit ønske, men AI'en har tekniske begrænsninger.
            </p>
          )}
        </div>
      )}
    </div>
  );
}

// /server/routes.ts

function buildPrompt(
  roomType: string, 
  style: string, 
  tier: string, 
  budget: number,
  customNotes?: string
) {
  const styleConfig = styleVocabulary[style];
  const tierConfig = styleConfig[tier];
  
  // Byg basis prompt
  let prompt = `
    Transform this ${roomType} to ${style} style.
    ${tierConfig.prompt}
    
    Style elements: ${styleConfig.colors.join(", ")}.
    Materials: ${styleConfig.materials.join(", ")}.
    
    Maintain exact room structure, windows, doors.
    NO layout changes.
  `;
  
  // Tilføj brugerønsker med HØJ prioritet
  if (customNotes && customNotes.trim().length > 0) {
    prompt += `
      
      ╔══════════════════════════════════════╗
      ║  USER REQUEST - HIGH PRIORITY        ║
      ║  ${customNotes}                      ║
      ╚══════════════════════════════════════╝
      
      CRITICAL: Incorporate the above user request as much as possible.
      Prioritize specific elements mentioned (colors, furniture, style).
      If request conflicts with default style, favor user request.
    `;
  }
  
  // Ekstra instruktion for specifikke ønsker
  if (customNotes?.toLowerCase().includes('væg') || 
      customNotes?.toLowerCase().includes('farve')) {
    prompt += `
      WALL COLOR PRIORITY: If user mentioned wall color, 
      ensure walls are EXACTLY that color, not default white.
    `;
  }
  
  return prompt.trim();
}
