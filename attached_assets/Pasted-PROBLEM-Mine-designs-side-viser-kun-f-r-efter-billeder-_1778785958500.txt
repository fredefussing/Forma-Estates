2. server/routes.ts — GET /api/designs endpoint

Change the query to JOIN product_matches:

// OLD: only designs
const designs = await pool.query('SELECT * FROM designs ORDER BY id DESC');

// NEW: include matches
const designs = await pool.query(`
  SELECT d.*, 
    COALESCE(
      json_agg(
        json_build_object(
          'id', p.id,
          'name', p.name,
          'name_en', p.name_en,
          'price', p.price,
          'currency', p.currency,
          'image_url', p.image_url,
          'affiliate_url', p.affiliate_url,
          'retailer', p.retailer,
          'match_type', pm.match_type,
          'match_score', pm.match_score,
          'rank', pm.rank
        ) ORDER BY pm.rank
      ) FILTER (WHERE p.id IS NOT NULL),
      '[]'::json
    ) as product_matches
  FROM designs d
  LEFT JOIN product_matches pm ON d.id = pm.design_id
  LEFT JOIN products p ON pm.product_id = p.id
  GROUP BY d.id
  ORDER BY d.id DESC
`);

// Return as array
for (const design of designs.rows) {
  design.product_matches = Array.isArray(design.product_matches) 
    ? design.product_matches 
    : JSON.parse(design.product_matches || '[]');
}
