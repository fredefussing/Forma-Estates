import { useQuery } from "@tanstack/react-query";
import { trpc } from "@/lib/trpc";
import { Card, CardContent } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Skeleton } from "@/components/ui/skeleton";
import { ShoppingBag, ExternalLink, Sparkles, Sofa } from "lucide-react";

interface ProductMatchesProps {
  designId: number;
}

interface ProductMatch {
  id: number;
  name: string;
  name_en: string | null;
  price: number | null;
  currency: string | null;
  image_url: string | null;
  affiliate_url: string | null;
  retailer: string;
  match_type: "same_style" | "alternative";
  match_score: number;
  rank: number;
}

export function ProductMatches({ designId }: ProductMatchesProps) {
  const { data, isLoading } = trpc.designs.getProducts.useQuery(
    { designId },
    { enabled: designId > 0 }
  );

  if (isLoading) return <ProductMatchesSkeleton />;

  if (!data?.products || data.products.length === 0) {
    return (
      <div className="flex flex-col items-center justify-center py-12 text-center">
        <Sofa className="h-10 w-10 text-stone-300 mb-3" />
        <p className="text-stone-400 text-sm">Finder produkter...</p>
        <p className="text-stone-300 text-xs mt-1">
          Det tager et øjeblik efter billedet er genereret.
        </p>
      </div>
    );
  }

  const sameStyle = data.products.filter(
    (p: ProductMatch) => p.match_type === "same_style"
  );
  const alternatives = data.products.filter(
    (p: ProductMatch) => p.match_type === "alternative"
  );

  return (
    <div className="space-y-6">
      {/* Header */}
      <div className="flex items-center gap-2">
        <ShoppingBag className="h-5 w-5 text-stone-600" />
        <h3 className="font-semibold text-stone-800">Shop stilen</h3>
        <Badge variant="secondary" className="ml-auto text-xs">
          {data.products.length} produkter
        </Badge>
      </div>

      {/* Info badge */}
      <div className="flex items-start gap-2 bg-amber-50 border border-amber-200 rounded-lg p-3">
        <Sparkles className="h-4 w-4 text-amber-600 mt-0.5 flex-shrink-0" />
        <p className="text-xs text-amber-800">
          <strong>Inspiration, ikke identitet.</strong> AI-genererede møbler er
          fiktive. Produkterne matcher stilen og farven — men er ikke 1:1 kopier.
        </p>
      </div>

      {/* "I samme stil" */}
      {sameStyle.length > 0 && (
        <div className="space-y-3">
          <h4 className="text-sm font-medium text-emerald-700 flex items-center gap-1.5">
            <span className="w-2 h-2 rounded-full bg-emerald-500" />
            I samme stil
          </h4>
          <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
            {sameStyle.map((product: ProductMatch) => (
              <ProductCard key={product.id} product={product} />
            ))}
          </div>
        </div>
      )}

      {/* "Alternativ" */}
      {alternatives.length > 0 && (
        <div className="space-y-3">
          <h4 className="text-sm font-medium text-stone-500 flex items-center gap-1.5">
            <span className="w-2 h-2 rounded-full bg-stone-400" />
            Alternativ
          </h4>
          <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
            {alternatives.map((product: ProductMatch) => (
              <ProductCard key={product.id} product={product} />
            ))}
          </div>
        </div>
      )}
    </div>
  );
}

function ProductCard({ product }: { product: ProductMatch }) {
  const displayName = product.name_en || product.name;
  const priceDisplay = product.price
    ? `${product.price.toLocaleString("da-DK")} ${product.currency || "DKK"}`
    : "Se pris";

  return (
    <Card className="border-stone-200 overflow-hidden hover:shadow-md transition-shadow group">
      <div className="aspect-square bg-stone-100 relative overflow-hidden">
        {product.image_url ? (
          <img
            src={product.image_url}
            alt={displayName}
            className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
          />
        ) : (
          <div className="w-full h-full flex items-center justify-center">
            <Sofa className="h-12 w-12 text-stone-300" />
          </div>
        )}
        <Badge
          className={`absolute top-2 right-2 text-xs ${
            product.match_type === "same_style"
              ? "bg-emerald-500 text-white"
              : "bg-stone-500 text-white"
          }`}
        >
          {product.match_type === "same_style" ? "I samme stil" : "Alternativ"}
        </Badge>
      </div>

      <CardContent className="p-3 space-y-2">
        <p className="text-sm font-medium text-stone-800 line-clamp-2 group-hover:text-emerald-700 transition-colors">
          {displayName}
        </p>

        <div className="flex items-center justify-between">
          <p className="text-sm font-semibold text-stone-900">{priceDisplay}</p>
          <span className="text-xs text-stone-400">{product.retailer}</span>
        </div>

        {product.affiliate_url ? (
          <a
            href={product.affiliate_url}
            target="_blank"
            rel="noopener noreferrer"
            className="block"
          >
            <Button
              size="sm"
              variant="outline"
              className="w-full text-xs border-emerald-200 text-emerald-700 hover:bg-emerald-50"
            >
              <ExternalLink className="h-3 w-3 mr-1" />
              Se produkt
            </Button>
          </a>
        ) : (
          <Button
            size="sm"
            variant="outline"
            disabled
            className="w-full text-xs opacity-50"
          >
            Intet link
          </Button>
        )}
      </CardContent>
    </Card>
  );
}

function ProductMatchesSkeleton() {
  return (
    <div className="space-y-6">
      <div className="flex items-center gap-2">
        <Skeleton className="h-5 w-5" />
        <Skeleton className="h-5 w-32" />
        <Skeleton className="h-5 w-16 ml-auto" />
      </div>
      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
        {[1, 2, 3].map((i) => (
          <Card key={i} className="border-stone-200 overflow-hidden">
            <Skeleton className="aspect-square" />
            <CardContent className="p-3 space-y-2">
              <Skeleton className="h-4 w-full" />
              <Skeleton className="h-4 w-20" />
              <Skeleton className="h-8 w-full" />
            </CardContent>
          </Card>
        ))}
      </div>
    </div>
  );
}

// GET /api/designs/:id/products — returner matchede produkter
app.get("/api/designs/:id/products", async (req, res) => {
  const designId = parseInt(req.params.id);
  
  const result = await pool.query(`
    SELECT p.id, p.name, p.name_en, p.price, p.currency,
           p.image_url, p.affiliate_url, p.retailer,
           pm.match_type, pm.match_score, pm.rank
    FROM product_matches pm
    JOIN products p ON pm.product_id = p.id
    WHERE pm.design_id = $1
    ORDER BY pm.rank
  `, [designId]);

  res.json({ 
    products: result.rows,
    count: result.rows.length 
  });
});
