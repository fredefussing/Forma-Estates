import { useState, useEffect } from 'react';

interface StyleProduct {
  id: string;
  name: string;
  image_url: string;
  affiliate_link: string;
  price: number | null;
  tags: Record<string, any>;
  tag_confidence: number;
  match_score: number;
}

interface Props {
  style: string;
  roomType: string;
  budget?: 'budget' | 'standard' | 'luxury';
  aiImageUrl?: string; // Det genererede AI billede
}

export function ShopThisStyle({ 
  style, 
  roomType, 
  budget = 'standard',
  aiImageUrl 
}: Props) {
  const [products, setProducts] = useState<StyleProduct[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [activeProduct, setActiveProduct] = useState<string | null>(null);

  useEffect(() => {
    fetchProducts();
  }, [style, roomType, budget]);

  async function fetchProducts() {
    try {
      setLoading(true);
      const res = await fetch(
        `/api/style-products?style=${style}&room=${roomType}&budget=${budget}&limit=8`
      );
      const data = await res.json();

      if (data.success) {
        setProducts(data.products);
      } else {
        setError(data.message || "Ingen produkter fundet");
      }
    } catch (e) {
      setError("Kunne ikke hente produkter");
    } finally {
      setLoading(false);
    }
  }

  if (loading) {
    return (
      <div className="mt-8 p-8 bg-gray-50 rounded-xl animate-pulse">
        <div className="h-6 bg-gray-200 rounded w-1/3 mb-4" />
        <div className="grid grid-cols-4 gap-4">
          {[1,2,3,4].map(i => (
            <div key={i} className="h-48 bg-gray-200 rounded" />
          ))}
        </div>
      </div>
    );
  }

  if (error && products.length === 0) {
    return (
      <div className="mt-8 p-6 bg-orange-50 border border-orange-200 rounded-xl">
        <p className="text-orange-800">{error}</p>
        <button 
          onClick={fetchProducts}
          className="mt-2 text-sm text-orange-600 underline hover:text-orange-800"
        >
          Prøv igen
        </button>
      </div>
    );
  }

  return (
    <div className="mt-8 space-y-6">
      {/* Header */}
      <div className="flex items-center justify-between">
        <div>
          <h2 className="text-2xl font-bold text-gray-900">
            🛍️ Shop dette look
          </h2>
          <p className="text-gray-500 mt-1">
            {style.charAt(0).toUpperCase() + style.slice(1)} · {getRoomLabel(roomType)}
            {products.length > 0 && ` · ${products.length} produkter`}
          </p>
        </div>

        {/* Budget toggle */}
        <div className="flex bg-gray-100 rounded-lg p-1">
          {(['budget', 'standard', 'luxury'] as const).map(b => (
            <button
              key={b}
              onClick={() => b !== budget && fetchProducts()} // Note: would need state lift
              className={`px-3 py-1.5 text-sm rounded-md transition ${
                b === budget 
                  ? 'bg-white text-gray-900 shadow-sm' 
                  : 'text-gray-500 hover:text-gray-700'
              }`}
            >
              {b === 'budget' ? 'Budget' : b === 'standard' ? 'Standard' : 'Luksus'}
            </button>
          ))}
        </div>
      </div>

      {/* AI Billede reference */}
      {aiImageUrl && (
        <div className="relative rounded-xl overflow-hidden bg-gray-100 aspect-video">
          <img 
            src={aiImageUrl} 
            alt="Dit AI design" 
            className="w-full h-full object-cover opacity-50"
          />
          <div className="absolute inset-0 flex items-center justify-center">
            <p className="bg-black/70 text-white px-4 py-2 rounded-lg text-sm">
              Produkterne nedenfor matcher stilen på dit billede
            </p>
          </div>
        </div>
      )}

      {/* Produkt grid */}
      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
        {products.map((product, index) => (
          <ProductCard 
            key={product.id}
            product={product}
            index={index}
            isActive={activeProduct === product.id}
            onHover={() => setActiveProduct(product.id)}
            onLeave={() => setActiveProduct(null)}
          />
        ))}
      </div>

      {/* Footer info */}
      <div className="text-center py-4 border-t border-gray-100">
        <p className="text-sm text-gray-500">
          Alle produkter er håndplukket til {style}-stilen · 
          <span className="text-blue-600"> Affiliate links — du støtter os når du shopper</span>
        </p>
      </div>
    </div>
  );
}

function ProductCard({ 
  product, 
  index,
  isActive,
  onHover,
  onLeave
}: { 
  product: StyleProduct; 
  index: number;
  isActive: boolean;
  onHover: () => void;
  onLeave: () => void;
}) {
  return (
    <a
      href={product.affiliate_link}
      target="_blank"
      rel="noopener noreferrer"
      className={`group block rounded-xl border transition-all duration-300 ${
        isActive 
          ? 'border-blue-400 shadow-lg scale-[1.02]' 
          : 'border-gray-200 hover:border-blue-300 hover:shadow-md'
      }`}
      onMouseEnter={onHover}
      onMouseLeave={onLeave}
    >
      {/* Billede */}
      <div className="relative aspect-square bg-gray-100 rounded-t-xl overflow-hidden">
        <img 
          src={product.image_url} 
          alt={product.name}
          className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
          loading="lazy"
        />

        {/* Badge */}
        <div className="absolute top-3 left-3 flex gap-2">
          <span className="bg-blue-600 text-white text-xs px-2.5 py-1 rounded-full font-medium">
            #{index + 1}
          </span>
          {product.tag_confidence > 0.85 && (
            <span className="bg-green-500 text-white text-xs px-2 py-1 rounded-full">
              Top match
            </span>
          )}
        </div>

        {/* Tags overlay */}
        <div className="absolute bottom-0 left-0 right-0 bg-gradient-to-t from-black/70 to-transparent p-3">
          <div className="flex gap-1 flex-wrap">
            <TagBadge label={product.tags?.color} />
            <TagBadge label={product.tags?.material} />
            <TagBadge label={product.tags?.type} />
          </div>
        </div>
      </div>

      {/* Info */}
      <div className="p-4">
        <h3 className="font-medium text-gray-900 line-clamp-2 group-hover:text-blue-600 transition-colors">
          {product.name}
        </h3>

        {product.price && (
          <p className="text-lg font-bold text-gray-900 mt-2">
            {product.price.toLocaleString('da-DK')} kr
          </p>
        )}

        <div className="mt-3 flex items-center gap-1 text-sm text-blue-600 font-medium">
          <span>Se produkt</span>
          <svg className="w-4 h-4 group-hover:translate-x-1 transition-transform" fill="none" stroke="currentColor" viewBox="0 0 24 24">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 5l7 7-7 7" />
          </svg>
        </div>
      </div>
    </a>
  );
}

function TagBadge({ label }: { label?: string }) {
  if (!label || label === 'unknown') return null;
  return (
    <span className="bg-white/90 text-gray-700 text-xs px-2 py-0.5 rounded">
      {label}
    </span>
  );
}

function getRoomLabel(roomType: string): string {
  const labels: Record<string, string> = {
    living_room: 'Stue',
    bedroom: 'Soveværelse',
    kitchen: 'Køkken',
    bathroom: 'Badeværelse',
    dining_room: 'Spisestue',
    office: 'Kontor',
    hallway: 'Entré'
  };
  return labels[roomType] || roomType;
}
