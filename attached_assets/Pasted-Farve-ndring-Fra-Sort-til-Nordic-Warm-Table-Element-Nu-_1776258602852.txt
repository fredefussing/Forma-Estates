Komplet Farvepalette
css
Copy
:root {
  --bg-primary: #FAF9F7;        /* Varm hvid — hovedbaggrund */
  --bg-secondary: #FFFFFF;      /* Ren hvid — cards, badges */
  --text-primary: #1A1A1A;      /* Næsten sort — overskrifter */
  --text-secondary: #6B7280;    /* Grå — subtext, labels */
  --accent: #C4A77D;            /* Varm guld — hover, indicators */
  --cta-bg: #1A1A1A;            /* Mørk — primær knap */
  --cta-text: #FFFFFF;          /* Hvid — knaptekst */
  --badge-bg: rgba(26, 26, 26, 0.8); /* FØR/EFTER badges */
}
