async function sendVstWorkflow(originalImageUrl, roomType, style) {
  // STEP 1: Generer tomt rum
  const emptyForm = new FormData();
  emptyForm.append("uploadUrl", originalImageUrl);
  
  const emptyRes = await fetch(
    "https://api.collov.ai/flair/enterpriseApi/vst/generateEmptyRoom",
    {
      method: "POST",
      headers: { apiKey: process.env.COLOV_API_KEY },
      body: emptyForm
    }
  );
  const emptyData = await emptyRes.json();
  const emptyTaskId = emptyData.data.id;
  
  // Poll empty room
  const emptyRoomUrl = await pollEmptyRoom(emptyTaskId);
  
  // STEP 2: Stage med BEGGE URLs
  const stageForm = new FormData();
  stageForm.append("uploadUrl", originalImageUrl);    // ← ORIGINAL foto
  stageForm.append("emptyRoomUrl", emptyRoomUrl);     // ← Tomt rum
  stageForm.append("roomType", roomType.toLowerCase());
  stageForm.append("style", style.toLowerCase());
  
  const stageRes = await fetch(
    "https://api.collov.ai/flair/enterpriseApi/vst/generateImgOnCommon",
    {
      method: "POST",
      headers: { apiKey: process.env.COLOV_API_KEY },
      body: stageForm
    }
  );
  const stageData = await stageRes.json();
  return stageData.data.uuid;  // ← UUID til polling
}

3. POLL VST — korrekt parsing:

async function pollVstResult(uuid) {
  const form = new FormData();
  form.append("uuid", uuid);
  
  const res = await fetch(
    "https://api.collov.ai/flair/enterpriseApi/vst/getRecord",
    {
      method: "GET",  // ← GET, multipart body
      headers: { apiKey: process.env.COLOV_API_KEY },
      body: form
    }
  );
  
  const data = await res.json();
  
  // KORREKT PATH: generateRecordList[0]
  const record = data.data?.generateRecordList?.[0];
  if (!record) return { status: "pending" };
  
  return {
    status: record.status.toLowerCase(),  // "SUCCESS" | "PENDING" | "FAILED"
    imageUrl: record.generateUrl,         // ← billed URL
    duration: record.durationTime
  };
}

4. SHARP.JS — aggressiv:

const sharp = require("sharp");

async function enhanceVstImage(buffer) {
  return sharp(buffer)
    // KANTER — maksimal skarphed
    .sharpen({ 
      sigma: 1.5,      // Radius
      flat: 0.3,       // Mindre noise-reduktion = flere detaljer
      jagged: 3        // Maksimal kant-forstærkning
    })
    // LOKAL KONTRAST — detaljer synlige
    .clahe({ 
      width: 50, 
      height: 50, 
      maxSlope: 3 
    })
    // MIKRO-SKARPHE
    .unsharpMask({ 
      sigma: 2, 
      flat: 0.5, 
      jagged: 2 
    })
    // FARVE — lidt mere liv
    .modulate({ 
      saturation: 1.05, 
      brightness: 1.02 
    })
    .toBuffer();
}
