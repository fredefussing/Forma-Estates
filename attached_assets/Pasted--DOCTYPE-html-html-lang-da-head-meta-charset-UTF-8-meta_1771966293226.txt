        // State
        let currentStep = 1;
        let selectedRoom = null;
        let selectedStyle = null;
        let selectedBudget = null;

        // Initialize
        document.addEventListener('DOMContentLoaded', () => {
            renderRooms();
            renderStyles();
        });

        function showQuiz() {
            document.getElementById('quizContainer').classList.add('active');
            document.getElementById('quizContainer').scrollIntoView({ behavior: 'smooth' });
        }

        function renderRooms() {
            const grid = document.getElementById('roomGrid');
            grid.innerHTML = rooms.map(room => `
                <div class="room-card" onclick="selectRoom('${room.id}', '${room.name}')" data-room="${room.id}">
                    <div class="room-icon">${room.icon}</div>
                    <div>${room.name}</div>
                </div>
            `).join('');
        }

        function renderStyles() {
            const grid = document.getElementById('styleGrid');
            grid.innerHTML = Object.entries(styles).map(([key, style]) => `
                <div class="style-card" onclick="selectStyle('${key}')" data-style="${key}">
                    ${style.hasImages ? 
                        `<div class="style-image-placeholder">📷</div>` : 
                        `<div class="style-image-placeholder">${getStyleEmoji(key)}</div>`
                    }
                    <div class="style-info">
                        <h4>${style.name}</h4>
                        <p>${style.description}</p>
                    </div>
                </div>
            `).join('');
        }

        function getStyleEmoji(style) {
            const emojis = {
                badboy: '🖤',
                luksus: '✨',
                industriel: '🏭',
                boheme: '🎨',
                minimalistisk: '○',
                klassisk: '🏛️'
            };
            return emojis[style] || '🏠';
        }

        function selectRoom(roomId, roomName) {
            selectedRoom = { id: roomId, name: roomName };
            
            // Update UI
            document.querySelectorAll('.room-card').forEach(card => {
                card.classList.remove('selected');
                if(card.dataset.room === roomId) card.classList.add('selected');
            });

            // Update preview header
            document.getElementById('previewRoomName').textContent = roomName;

            // Go to next step after short delay
            setTimeout(() => {
                goToStep(2);
            }, 300);
        }

        function selectStyle(styleKey) {
            selectedStyle = styleKey;
            const style = styles[styleKey];
            
            // Update UI
            document.querySelectorAll('.style-card').forEach(card => {
                card.classList.remove('selected');
                if(card.dataset.style === styleKey) card.classList.add('selected');
            });

            // Update preview
            const previewContainer = document.getElementById('previewContainer');
            const previewText = document.getElementById('previewStyleText');
            
            if (style.hasImages && style.images[selectedRoom.id]) {
                previewContainer.innerHTML = `<img src="${style.images[selectedRoom.id]}" class="preview-image" alt="${style.name}">`;
                previewText.textContent = `${style.name} stil i dit ${selectedRoom.name.toLowerCase()} - perfekt match!`;
            } else {
                previewContainer.innerHTML = `
                    <div class="preview-placeholder" style="background: ${getStyleColor(styleKey)}; color: white; flex-direction: column;">
                        <span style="font-size: 3rem; margin-bottom: 1rem;">${getStyleEmoji(styleKey)}</span>
                        <span>${style.name}</span>
                    </div>
                `;
                previewText.textContent = `${style.name} stil - ${style.description}. Tilføj dit billede for at se transformationen!`;
            }

            // Go to next step after delay
            setTimeout(() => {
                goToStep(3);
            }, 800);
        }

        function getStyleColor(style) {
            const colors = {
                badboy: '#2c2c2c',
                luksus: '#d4af37',
                industriel: '#5a5a5a',
                boheme: '#e74c3c',
                minimalistisk: '#ecf0f1',
                klassisk: '#34495e'
            };
            return colors[style] || '#95a5a6';
        }

        function selectBudget(budgetValue, budgetLabel) {
            selectedBudget = { value: budgetValue, label: budgetLabel };
            
            // Update UI
            document.querySelectorAll('.budget-card').forEach(card => {
                card.classList.remove('selected');
            });
            event.currentTarget.classList.add('selected');

            // Update preview
            const style = styles[selectedStyle];
            const previewImg = document.getElementById('budgetPreviewImage');
            const previewPlaceholder = document.getElementById('budgetPreviewPlaceholder');
            
            if (style.hasImages && style.images[selectedRoom.id]) {
                previewImg.src = style.images[selectedRoom.id];
                previewImg.style.display = 'block';
                previewPlaceholder.style.display = 'none';
            } else {
                previewImg.style.display = 'none';
                previewPlaceholder.style.display = 'flex';
                previewPlaceholder.style.background = getStyleColor(selectedStyle);
                previewPlaceholder.style.color = 'white';
                previewPlaceholder.innerHTML = `<span style="font-size: 2rem;">${getStyleEmoji(selectedStyle)} ${style.name}</span>`;
            }

            document.getElementById('budgetSummary').innerHTML = `
                <strong>${selectedRoom.name}</strong> + 
                <strong>${style.name}</strong> + 
                <strong>${budgetLabel}</strong>
            `;

            // Show result after delay
            setTimeout(() => {
                showResult();
            }, 600);
        }

        function goToStep(step) {
            // Hide all steps
            document.querySelectorAll('.quiz-step').forEach(s => s.classList.remove('active'));
            
            // Show target step
            document.getElementById(`step${step}`).classList.add('active');
            
            // Update progress
            updateProgress(step);
            
            currentStep = step;
        }

        function updateProgress(step) {
            for (let i = 1; i <= 3; i++) {
                const indicator = document.getElementById(`step${i}-indicator`);
                indicator.classList.remove('active', 'completed');
                if (i < step) indicator.classList.add('completed');
                else if (i === step) indicator.classList.add('active');
            }
        }

        function showResult() {
            const style = styles[selectedStyle];
            const roomName = selectedRoom.name.toLowerCase();
            const budgetLabel = selectedBudget.label;
            
            // Hide quiz steps, show result
            document.querySelectorAll('.quiz-step').forEach(s => s.classList.remove('active'));
            document.getElementById('result').classList.add('active');
            
            // Hide progress bar
            document.querySelector('.progress-bar').style.display = 'none';
            
            // Set result image
            const resultImg = document.getElementById('resultImage');
            if (style.hasImages && style.images[selectedRoom.id]) {
                resultImg.src = style.images[selectedRoom.id];
                resultImg.style.display = 'block';
            } else {
                resultImg.style.display = 'none';
                // Add colored background instead
                const resultContainer = document.querySelector('.result-container');
                resultContainer.style.background = getStyleColor(selectedStyle);
                resultContainer.style.color = 'white';
            }
            
            // Set texts
            document.getElementById('resultStyle').textContent = style.name;
            
            const whyText = style.why
                .replace('{room}', roomName)
                .replace('{budget}', budgetLabel)
                .replace('{personality}', getPersonalityText(selectedStyle));
            
            document.getElementById('resultText').innerHTML = `
                ${whyText}<br><br>
                ${style.description}
            `;
            
            // Set features
            const featuresList = document.getElementById('resultFeatures');
            featuresList.innerHTML = style.features.map(f => `<li>${f}</li>`).join('');
            
            // Set price
            const priceRange = style.priceRanges[selectedBudget.value];
            document.getElementById('resultPrice').innerHTML = `
                💰 Prisniveau: ${priceRange} for komplet ${roomName}
            `;
        }

        function getPersonalityText(style) {
            const personalities = {
                skandinavisk: 'varme',
                moderne: 'strukturerede',
                badboy: 'selvsikre',
                luksus: 'kræsne',
                industriel: 'ægte',
                boheme: 'kreative',
                minimalistisk: 'fokuserede',
                klassisk: 'traditionelle'
            };
            return personalities[style] || 'unikke';
        }

            alert(`Videregår til upload med ${selectedStyle} stil pre-valgt til dit ${selectedRoom.name.toLowerCase()}!`);

        function resetQuiz() {
            // Reset state
            selectedRoom = null;
            selectedStyle = null;
            selectedBudget = null;
            
            // Reset UI
            document.querySelectorAll('.room-card, .style-card, .budget-card').forEach(c => {
                c.classList.remove('selected');
            });
            
            document.getElementById('previewContainer').innerHTML = '<div class="preview-placeholder">Vælg en stil for at se preview</div>';
            document.getElementById('previewStyleText').textContent = 'Vælg en stil fra listen...';
            document.getElementById('budgetPreviewImage').style.display = 'none';
            document.getElementById('budgetPreviewPlaceholder').style.display = 'flex';
            document.getElementById('budgetSummary').textContent = '';
            
            // Show progress bar again
            document.querySelector('.progress-bar').style.display = 'flex';
            
            // Go to step 1
            goToStep(1);
        }
