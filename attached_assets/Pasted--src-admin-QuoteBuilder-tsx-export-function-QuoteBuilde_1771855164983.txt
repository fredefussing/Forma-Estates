// /src/admin/QuoteBuilder.tsx

export function QuoteBuilder({ design }) {
  const [products, setProducts] = useState([]);
  const [totalPrice, setTotalPrice] = useState(0);
  
  // Admin tilføjer produkter manuelt
  const addProduct = () => {
    setProducts([...products, {
      name: '',
      retailer: '',
      price: 0,
      link: ''
    }]);
  };
  
  const updateProduct = (index, field, value) => {
    const updated = [...products];
    updated[index][field] = value;
    setProducts(updated);
    
    // Beregn total
    const total = updated.reduce((sum, p) => sum + (Number(p.price) || 0), 0);
    setTotalPrice(total);
  };
  
  const sendQuote = async () => {
    await fetch('/api/quotes', {
      method: 'POST',
      body: JSON.stringify({
        designId: design.id,
        products,
        totalPrice,
        margin: totalPrice * 0.25, // 25% markup
        finalPrice: totalPrice * 1.25
      })
    });
  };
  
  return (
    <div className="quote-builder">
      <h2>Byg tilbud til kunde</h2>
      
      <div className="design-preview">
        <img src={design.resultUrl} alt="AI genereret" />
        <p>Budget: {design.budget} kr ({design.tier})</p>
        <p>Stil: {design.style}</p>
      </div>
      
      <h3>Produkter</h3>
      {products.map((p, i) => (
        <div key={i} className="product-row">
          <input 
            placeholder="Produktnavn" 
            value={p.name}
            onChange={(e) => updateProduct(i, 'name', e.target.value)}
          />
          <select 
            value={p.retailer}
            onChange={(e) => updateProduct(i, 'retailer', e.target.value)}
          >
            <option value="">Vælg butik</option>
            <option value="IKEA">IKEA</option>
            <option value="Ilva">Ilva</option>
            <option value="Bolia">Bolia</option>
            <option value="Hay">Hay</option>
          </select>
          <input 
            type="number"
            placeholder="Pris" 
            value={p.price}
            onChange={(e) => updateProduct(i, 'price', e.target.value)}
          />
          <input 
            placeholder="Link" 
            value={p.link}
            onChange={(e) => updateProduct(i, 'link', e.target.value)}
          />
        </div>
      ))}
      
      <button onClick={addProduct}>+ Tilføj produkt</button>
      
      <div className="summary">
        <p>Indkøbspris: {totalPrice.toLocaleString()} kr</p>
        <p>Jeres margin (25%): {(totalPrice * 0.25).toLocaleString()} kr</p>
        <p><strong>Tilbud til kunde: {(totalPrice * 1.25).toLocaleString()} kr</strong></p>
      </div>
      
      <button onClick={sendQuote} className="send-quote">
        Send tilbud til kunde
      </button>
    </div>
  );
}
