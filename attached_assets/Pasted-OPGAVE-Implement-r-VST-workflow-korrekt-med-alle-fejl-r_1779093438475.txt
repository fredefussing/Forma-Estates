1. pollEmptyRoom — rettet felt-check:

async function pollEmptyRoom(taskId) {
  const maxAttempts = 20;
  const interval = 2000;
  
  for (let i = 0; i < maxAttempts; i++) {
    const res = await fetch(
      `https://api.collov.ai/flair/enterpriseApi/vst/getEmptyRoomRecord?id=${taskId}`,
      { headers: { apiKey: process.env.COLOV_API_KEY } }
    );
    const data = await res.json();
    const status = data.data?.status;
    
    if (status === "SUCCESS" && data.data?.emptyRoomUrl) {
      return data.data.emptyRoomUrl;
    }
    if (status === "FAILED") throw new Error("Empty room failed");
    
    await new Promise(r => setTimeout(r, interval));
  }
  throw new Error("Empty room timeout");
}

2. Sharp — uden unsharpMask, med dobbelt sharpen:

async function enhanceVstImage(buffer) {
  return sharp(buffer)
    .sharpen({ sigma: 1.5, flat: 0.3, jagged: 3 })
    .sharpen({ sigma: 0.5, flat: 2, jagged: 0.5 })
    .clahe({ width: 50, height: 50, maxSlope: 3 })
    .sharpen({ sigma: 0.3, flat: 0.5, jagged: 1.5 })
    .jpeg({ quality: 95, mozjpeg: true })
    .toBuffer();
}

3. VST workflow — korrekte parametre:

async function runVstDesign(originalUrl, roomType, style) {
  const styleLower = style.toLowerCase();
  
  // Step 1: Empty room
  const emptyForm = new FormData();
  emptyForm.append("uploadUrl", originalUrl);
  
  const emptyRes = await fetch(
    "https://api.collov.ai/flair/enterpriseApi/vst/generateEmptyRoom",
    {
      method: "POST",
      headers: { apiKey: process.env.COLOV_API_KEY },
      body: emptyForm
    }
  );
  const emptyData = await emptyRes.json();
  const emptyRoomUrl = await pollEmptyRoom(emptyData.data.id);
  
  // Step 2: Stage — uploadUrl=original, emptyRoomUrl=tomt rum
  const stageForm = new FormData();
  stageForm.append("uploadUrl", originalUrl);
  stageForm.append("emptyRoomUrl", emptyRoomUrl);
  stageForm.append("roomType", styleLower);
  stageForm.append("style", styleLower);
  
  const stageRes = await fetch(
    "https://api.collov.ai/flair/enterpriseApi/vst/generateImgOnCommon",
    {
      method: "POST",
      headers: { apiKey: process.env.COLOV_API_KEY },
      body: stageForm
    }
  );
  const stageData = await stageRes.json();
  const uuid = stageData.data.uuid;
  
  // Poll design — 25 forsøg × 2s = 50s max
  for (let i = 0; i < 25; i++) {
    await new Promise(r => setTimeout(r, 2000));
    
    const pollForm = new FormData();
    pollForm.append("uuid", uuid);
    
    const pollRes = await fetch(
      "https://api.collov.ai/flair/enterpriseApi/vst/getRecord",
      {
        method: "GET",
        headers: { apiKey: process.env.COLOV_API_KEY },
        body: pollForm
      }
    );
    const pollData = await pollRes.json();
    const record = pollData.data?.generateRecordList?.[0];
    
    if (record?.status === "SUCCESS" && record?.generateUrl) {
      // Hent billede
      const imageRes = await fetch(record.generateUrl);
      const imageBuffer = await imageRes.arrayBuffer();
      
      // Sharp
      const enhanced = await enhanceVstImage(Buffer.from(imageBuffer));
      
      // Gem
      const savedPath = await saveImage(enhanced, uuid);
      return { imageUrl: savedPath, uuid };
    }
    if (record?.status === "FAILED") throw new Error("Design failed");
  }
  
  throw new Error("Design timeout");
}

async function runDesignWorkflow(originalUrl, roomType, style) {
  const mode = process.env.SOLID10_MODE || 'none';
  
  if (mode === 'all') {
    return runSolid10Design(originalUrl, roomType, style);
  }
  
  if (mode === 'skandi' && style.toLowerCase() !== 'scandinavian') {
    return runSolid10Design(originalUrl, roomType, style);
  }
  
  // Prøv VST
  try {
    const result = await Promise.race([
      runVstDesign(originalUrl, roomType, style),
      new Promise((_, reject) => 
        setTimeout(() => reject(new Error('VST_TIMEOUT')), 90000)
      )
    ]);
    return result;
  } catch (err) {
    console.log('[Workflow] VST failed:', err.message, '→ SOLID10 fallback');
    return runSolid10Design(originalUrl, roomType, style);
  }
}
