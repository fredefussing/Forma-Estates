import { pipeline } from "@xenova/transformers";

export interface Detection {
  box: [number, number, number, number];  // x1, y1, x2, y2 (normalized 0-1)
  label: string;
  score: number;
}

export interface GroupedDetection {
  box: [number, number, number, number];
  label: string;
  score: number;
  count: number;  // hvor mange detektioner er grupperet her
  isPrimary: boolean;
}

let detector: any = null;

export async function detectObjects(imageBuffer: Buffer): Promise<GroupedDetection[]> {
  // Lazy load model
  if (!detector) {
    detector = await pipeline(
      "object-detection",
      "Xenova/yolov8n-worldv2",  // eller din nuværende model
      { device: "cpu" }
    );
  }

  const rawDetections = await detector(imageBuffer, {
    threshold: 0.45,  // Lav threshold — vi filtrerer selv
    percentage: true  // Returner 0-1 koordinater
  });

  // Konverter til vores format
  const detections: Detection[] = rawDetections.map((d: any) => ({
    box: [d.box.xmin, d.box.ymin, d.box.xmax, d.box.ymax],
    label: d.label,
    score: d.score
  }));

  // GRUPPÉR og filtrér
  const grouped = groupDetections(detections, {
    maxDots: 8,
    confidenceThreshold: 0.65,  // Højere end før (0.55) for bedre kvalitet
    mergeThreshold: 0.25        // 25% overlap = samme møbel
  });

  return grouped;
}

interface GroupOptions {
  maxDots: number;
  confidenceThreshold: number;
  mergeThreshold: number;
}

export function groupDetections(
  detections: Detection[],
  options: GroupOptions
): GroupedDetection[] {
  const { maxDots, confidenceThreshold, mergeThreshold } = options;

  // STEP 1: Filtrér lav confidence
  let filtered = detections.filter(d => d.score >= confidenceThreshold);

  // STEP 2: Hvis for få, tillad lavere confidence
  if (filtered.length < 3) {
    const extra = detections
      .filter(d => d.score >= 0.45 && d.score < confidenceThreshold)
      .sort((a, b) => b.score - a.score)
      .slice(0, maxDots - filtered.length);
    filtered = [...filtered, ...extra];
  }

  // STEP 3: Sorter efter score (højeste først)
  filtered.sort((a, b) => b.score - b.score);

  // STEP 4: Gruppér overlappende detektioner
  const groups: GroupedDetection[] = [];

  for (const det of filtered) {
    let merged = false;

    for (const group of groups) {
      const iou = calculateIoU(det.box, group.box);

      if (iou > mergeThreshold) {
        // Udvid bounding box til at dække begge
        group.box = [
          Math.min(group.box[0], det.box[0]),
          Math.min(group.box[1], det.box[1]),
          Math.max(group.box[2], det.box[2]),
          Math.max(group.box[3], det.box[3])
        ];
        group.score = Math.max(group.score, det.score);
        group.count += 1;
        merged = true;
        break;
      }
    }

    if (!merged && groups.length < maxDots) {
      groups.push({
        box: det.box,
        label: det.label,
        score: det.score,
        count: 1,
        isPrimary: groups.length === 0  // Første er "primær"
      });
    }
  }

  // STEP 5: Smart label-korrektion
  // Hvis gruppe har >1 detektion med forskellige labels, vælg den mest specifikke
  for (const group of groups) {
    if (group.count > 1) {
      // Prioritet: møbler > dekoration > andet
      const furniturePriority: Record<string, number> = {
        'couch': 10, 'sofa': 10,
        'chair': 9, 'dining chair': 9, 'lounge chair': 9,
        'bed': 8,
        'dining table': 7, 'coffee table': 7, 'table': 7,
        'shelf': 6, 'cabinet': 6,
        'lamp': 5,
        'rug': 4,
        'potted plant': 2,
        'vase': 1, 'book': 1, 'clock': 1
      };

      // Behold label med højeste prioritet
      // (I praksis bruger vi den første, men her kan vi forbedre)
    }
  }

  // STEP 6: Fjern "støj" — små objekter der sandsynligvis er dekoration
  const minSize = 0.05;  // Mindst 5% af billedet
  const significant = groups.filter(g => {
    const width = g.box[2] - g.box[0];
    const height = g.box[3] - g.box[1];
    return (width * height) >= minSize;
  });

  // Hvis vi fjernede for mange, behold alle
  return significant.length >= 2 ? significant.slice(0, maxDots) : groups.slice(0, maxDots);
}

/**
 * Beregn Intersection over Union (IoU)
 */
function calculateIoU(boxA: [number, number, number, number], boxB: [number, number, number, number]): number {
  const xOverlap = Math.max(0, Math.min(boxA[2], boxB[2]) - Math.max(boxA[0], boxB[0]));
  const yOverlap = Math.max(0, Math.min(boxA[3], boxB[3]) - Math.max(boxA[1], boxB[1]));
  const overlapArea = xOverlap * yOverlap;

  const areaA = (boxA[2] - boxA[0]) * (boxA[3] - boxA[1]);
  const areaB = (boxB[2] - boxB[0]) * (boxB[3] - boxB[1]);

  if (areaA + areaB - overlapArea === 0) return 0;
  return overlapArea / (areaA + areaB - overlapArea);
}

/**
 * Konverter YOLO output til procent-baserede koordinater for frontend
 */
export function normalizeBox(
  box: [number, number, number, number],
  imageWidth: number,
  imageHeight: number
): [number, number, number, number] {
  return [
    (box[0] / imageWidth) * 100,
    (box[1] / imageHeight) * 100,
    (box[2] / imageWidth) * 100,
    (box[3] / imageHeight) * 100
  ];
}
