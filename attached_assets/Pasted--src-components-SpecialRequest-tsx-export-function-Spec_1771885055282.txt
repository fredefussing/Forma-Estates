// /src/components/SpecialRequest.tsx

export function SpecialRequest({ designId, imageUrl }) {
  const [showForm, setShowForm] = useState(false);
  const [request, setRequest] = useState('');
  const [submitted, setSubmitted] = useState(false);
  
  if (submitted) {
    return (
      <div className="special-request-confirmation">
        <h3>✓ Dit ønske er modtaget!</h3>
        <p>Vi kontakter dig indenfor 24 timer med et tilpasset design.</p>
        <p>Pris for manuel tilpasning: 500 kr</p>
      </div>
    );
  }
  
  return (
    <div className="special-request">
      <button 
        onClick={() => setShowForm(!showForm)}
        className="special-request-toggle"
      >
        Har du specielle ønsker vi ikke kan generere? (fx specifik vægfarve)
      </button>
      
      {showForm && (
        <div className="special-request-form">
          <textarea
            value={request}
            onChange={(e) => setRequest(e.target.value)}
            placeholder="Beskriv dit ønske - fx 'Mørkegrøn væg bag sengen'"
            rows={4}
          />
          
          <button 
            onClick={async () => {
              await fetch('/api/special-request', {
                method: 'POST',
                body: JSON.stringify({
                  designId,
                  originalImageUrl: imageUrl,
                  request,
                  price: 500
                })
              });
              setSubmitted(true);
            }}
            disabled={!request.trim()}
          >
            Send forespørgsel (500 kr)
          </button>
          
          <p className="note">
            Vi retter dit billede manuelt og sender indenfor 24 timer.
          </p>
        </div>
      )}
    </div>
  );
}

// /server/routes.ts - NY route

app.post('/api/special-request', async (req, res) => {
  const { designId, originalImageUrl, request, price } = req.body;
  
  // Gem i database
  await db.insert('specialRequests', {
    designId,
    originalImageUrl,
    request,
    price,
    status: 'pending',
    createdAt: new Date()
  });
  
  // Send email til DIG
  await sendEmail({
    to: 'DIN_EMAIL@gmail.com',
    subject: `NY MANUEL FORESPØRGSEL #${designId}`,
    html: `
      <h2>Ny kunde vil have manuel tilpasning</h2>
      
      <p><strong>Ønske:</strong> ${request}</p>
      <p><strong>Pris:</strong> ${price} kr</p>
      
      <h3>Originalt billede:</h3>
      <img src="${originalImageUrl}" style="max-width: 500px;" />
      
      <h3>Din opgave:</h3>
      <ol>
        <li>Åbn billedet i Photoshop/Canva</li>
        <li>Ret: ${request}</li>
        <li>Gem som ny version</li>
        <li>Upload til admin panel</li>
        <li>Send til kunde med faktura på ${price} kr</li>
      </ol>
      
      <a href="https://din-admin-url.dk/special/${designId}">
        Gå til admin
      </a>
    `
  });
  
  // Bekræft til kunde
  await sendEmail({
    to: req.user.email, // eller hent fra design
    subject: 'Dit specielle ønske er modtaget',
    html: `
      <p>Vi har modtaget dit ønske: "${request}"</p>
      <p>Vi arbejder på det og sender indenfor 24 timer.</p>
      <p>Pris: ${price} kr (betales ved modtagelse)</p>
    `
  });
  
  res.json({ success: true });
});
