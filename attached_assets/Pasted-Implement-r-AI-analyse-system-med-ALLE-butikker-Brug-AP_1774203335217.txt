export async function analyzeDesignImage(imageUrl: string, budget: number, roomType: string, style: string) {
  const response = await openai.chat.completions.create({
    model: "gpt-4o",
    messages: [
      {
        role: "user",
        content: [
          {
            type: "text",
            text: `Analyser dette ${roomType} i ${style} stil. Budget: ${budget} DKK.
            Fordel 85% på produkter (15% profit til mig).
            Identificer alle synlige møbler. Returner KUN JSON:
            {
              "products": [{"name": "Sofa", "description": "3-pers grå stof", "estimatedPrice": 8000}],
              "totalProductBudget": 25500,
              "profit": 4500
            }`
          },
          {
            type: "image_url",
            image_url: { url: imageUrl, detail: "high" }
          }
        ]
      }
    ],
    max_tokens: 2000,
    response_format: { type: "json_object" }
  });
  
  return JSON.parse(response.choices[0].message.content);
}

function getCategory(productName: string): string {
  const name = productName.toLowerCase();
  if (name.includes('seng')) return 'bed';
  if (name.includes('lampe') || name.includes('lys') || name.includes('pendel')) return 'lamp';
  if (name.includes('tæppe') || name.includes('gulvtæppe')) return 'rug';
  if (name.includes('pude') || name.includes('plaid') || name.includes('vase') || name.includes('dekoration')) return 'accessory';
  return 'furniture';
}

export function buildAllSearchLinks(productName: string, description: string, style: string) {
  const term = encodeURIComponent(`${productName} ${description} ${style}`);
  const category = getCategory(productName);
  return allStores[category] || allStores['furniture'];
}

export async function sendAIAnalysisEmail(to: string, design: any, analysis: any) {
  const html = `
    <h2>AI Analyse: ${design.roomType} i ${design.style}</h2>
    <p>Kunde: ${design.userEmail}</p>
    <p>Budget: ${design.budget} kr</p>
    <img src="${design.resultImageUrl}" style="max-width:400px"/>
    <h3>Produkter fundet af AI:</h3>
    <p><em>Vælg selv hvilken butik der har det der ligner mest og passer til prisen:</em></p>
    <table border="1" cellpadding="8" style="border-collapse: collapse;">
      <tr>
        <th>Produkt</th>
        <th>Beskrivelse</th>
        <th>Budget</th>
        <th>Butikker</th>
      </tr>
      ${analysis.products.map((p: any) => {
        const stores = buildAllSearchLinks(p.name, p.description, design.style);
        const term = encodeURIComponent(`${p.name} ${p.description} ${design.style}`);
        return `<tr>
          <td><strong>${p.name}</strong></td>
          <td>${p.description}</td>
          <td>${p.estimatedPrice} kr</td>
          <td>${stores.map(s => `<a href="${s.url}${term}" style="margin-right: 8px;">${s.name}</a>`).join('')}</td>
        </tr>`;
      }).join('')}
    </table>
    <p><strong>Total produkt budget:</strong> ${analysis.totalProductBudget} kr</p>
    <p><strong>Din profit (15%):</strong> ${analysis.profit} kr</p>
    <p><a href="${process.env.ADMIN_URL}/quotes/${design.id}">Byg tilbud i admin</a></p>
  `;
  
  await sendEmailWithBrevo({ to, subject: `AI Analyse: ${design.roomType} - ${design.budget} kr`, html });
}
